  1009→// POST handler - only for step=1 redirects
  1010→export async function POST(request: NextRequest) {
  1011→  const formData = await request.formData();
  1012→  const flow = formData.get('flow') as string;
  1013→  const step = formData.get('step') as string;
  1014→
  1015→  if (step === '1') {
  1016→    if (flow === 'dropoff') {
  1017→      return NextResponse.redirect(new URL('/kiosk-lite?action=dropoff', request.url), 303);
  1018→    }
  1019→    if (flow === 'pickup') {
  1020→      return NextResponse.redirect(new URL('/kiosk-lite?action=pickup', request.url), 303);
  1021→    }
  1022→  }
  1023→
  1024→  return NextResponse.redirect(new URL('/kiosk-lite', request.url), 303);
  1025→}
