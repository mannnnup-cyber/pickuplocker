   354→// Recent activity component
   355→function RecentActivity({ activities }: { activities: Array<{id: string; action: string; description: string | null; userName?: string; time: string}> }) {
   356→  const formatTime = (dateStr: string) => {
   357→    const date = new Date(dateStr)
   358→    const now = new Date()
   359→    const diff = Math.floor((now.getTime() - date.getTime()) / 1000 / 60)
   360→    if (diff < 60) return `${diff} mins ago`
   361→    if (diff < 1440) return `${Math.floor(diff / 60)} hours ago`
   362→    return `${Math.floor(diff / 1440)} days ago`
   363→  }
   364→
   365→  return (
   366→    <Card className="bg-white border-gray-200 shadow-sm text-gray-900">
   367→      <CardHeader>
   368→        <CardTitle className="text-[#111111] uppercase">Recent Activity</CardTitle>
   369→        <CardDescription className="text-gray-500">Latest events across all devices</CardDescription>
   370→      </CardHeader>
   371→      <CardContent>
   372→        {activities.length === 0 ? (
   373→          <p className="text-gray-500 text-center py-4">No recent activity</p>
   374→        ) : (
   375→          <div className="space-y-4">
   376→            {activities.map((activity) => (
   377→              <div key={activity.id} className="flex items-center gap-4">
   378→                <div className="rounded-full bg-[#FFD439] p-2">
   379→                  <Package className="h-4 w-4 text-[#111111]" />
   380→                </div>
   381→                <div className="flex-1 space-y-1">
   382→                  <p className="text-sm font-medium text-[#111111]">{activity.action.replace(/_/g, ' ')}</p>
   383→                  <p className="text-sm text-gray-500">{activity.description || activity.userName || ''}</p>
   384→                </div>
   385→                <div className="text-sm text-gray-500">{formatTime(activity.time)}</div>
   386→              </div>
   387→            ))}
   388→          </div>
   389→        )}
   390→      </CardContent>
   391→    </Card>
   392→  )
   393→}

   395→// System Status component
   396→function SystemStatusCard() {
   397→  const [status, setStatus] = React.useState<SystemStatus | null>(null)
   398→  const [loading, setLoading] = React.useState(true)
   399→
   400→  const fetchStatus = React.useCallback(async () => {
   401→    setLoading(true)
   402→    try {
   403→      const res = await fetch('/api/status')
   404→      const data = await res.json()
   405→      setStatus(data)
   406→    } catch (error) {
   407→      console.error('Failed to fetch status:', error)
   408→    } finally {
   409→      setLoading(false)
   410→    }
   411→  }, [])
   412→
   413→  React.useEffect(() => {
   414→    fetchStatus()
   415→    // Reduced polling: 60 seconds instead of 30 (API has ISR cache now)
   416→    const interval = setInterval(fetchStatus, 60000)
   417→    
   418→    // Also refresh when tab becomes visible
   419→    const handleVisibility = () => {
   420→      if (document.visibilityState === 'visible') {
   421→        fetchStatus()
   422→      }
   423→    }
   424→    document.addEventListener('visibilitychange', handleVisibility)
   425→    
   426→    return () => {
   427→      clearInterval(interval)
   428→      document.removeEventListener('visibilitychange', handleVisibility)
   429→    }
   430→  }, [fetchStatus])
   431→
   432→  const getStatusIcon = (statusType: string) => {
   433→    switch (statusType) {
   434→      case 'online': return <CheckCircle className="h-4 w-4 text-green-500" />

   438→    }
   439→  }

   450→  }
   451→
   452→  const getOverallStatusBadge = () => {

   455→      online: 'bg-green-500 text-white',

   461→        {status.overallStatus.toUpperCase()}

   463→    )
   464→  }
   465→
   466→  return (
   467→    <Card className="bg-white border-gray-200 shadow-sm text-gray-900">

   531→  const [syncResult, setSyncResult] = React.useState<{success: boolean; message: string} | null>(null)

   567→  React.useEffect(() => {

   581→      clearInterval(interval)

   584→  }, [fetchSyncData])

   586→  const formatTimeAgo = (dateStr: string | null) => {
   587→    if (!dateStr) return 'Never'
