   134→  // Auto-recovery: reload page if it's been stuck/idle too long (prevents white screen)

   136→    const HEARTBEAT_KEY = 'pickup_kiosk_heartbeat'
   137→    const RECOVERY_MS = 30 * 60 * 1000 // 30 minutes - if no heartbeat, something is wrong
   138→    
   139→    // Write heartbeat
   140→    const heartbeat = setInterval(() => {
   141→      try { localStorage.setItem(HEARTBEAT_KEY, Date.now().toString()) } catch {}
   142→    }, 60000)
   143→
   144→    // Check if previous session crashed (no heartbeat for 30 min = white screen)
   145→    try {
   146→      const lastHeartbeat = parseInt(localStorage.getItem(HEARTBEAT_KEY) || '0')
   147→      if (lastHeartbeat && (Date.now() - lastHeartbeat > RECOVERY_MS)) {
   148→        console.log('[Kiosk] Previous session appears to have crashed. Clearing state.')
   149→        localStorage.removeItem(HEARTBEAT_KEY)
   150→      }
   151→    } catch {}
   152→
   153→    // Write initial heartbeat
   154→    try { localStorage.setItem(HEARTBEAT_KEY, Date.now().toString()) } catch {}
   155→
   156→    return () => {
   157→      clearInterval(heartbeat)
   158→      try { localStorage.removeItem(HEARTBEAT_KEY) } catch {}
   159→    }

   319→          <div className="text-center text-4xl font-mono tracking-widest text-gray-800 min-h-[48px]">

   431→  // Poll payment status with smart backoff
   432→  const pollPaymentStatus = async (paymentId: string) => {
   433→    const maxAttempts = 45 // ~6 minutes with backoff
   434→    let attempts = 0
   435→
   436→    const poll = async () => {
   437→      // Stop if component unmounted
   438→      if (!isMountedRef.current) return
   439→
   440→      if (attempts >= maxAttempts) {
   441→        setError("Payment timeout. Please try again.")
   442→        return
   443→      }
   444→
   445→      // Stop polling if tab is hidden
   446→      if (document.visibilityState === 'hidden') {
   447→        // Resume when visible again - track listener for cleanup
   448→        const handleVisible = () => {
   449→          if (document.visibilityState === 'visible') {
   450→            document.removeEventListener('visibilitychange', handleVisible)
   451→            if (visibilityHandlerRef.current === handleVisible) {
   452→              visibilityHandlerRef.current = null
   453→            }
   454→            poll()
   455→          }
   456→        }
   457→        visibilityHandlerRef.current = handleVisible
   458→        document.addEventListener('visibilitychange', handleVisible)
   459→        return
   460→      }
   461→
   462→      attempts++
   463→
   464→      try {
   465→        const res = await fetch(`/api/kiosk/payment?paymentId=${paymentId}`)
   466→        const data = await res.json()
   467→
   468→        if (!isMountedRef.current) return
   469→
   470→        if (data.status === "completed") {
   471→          setCode(data.saveCode || "")
   472→          // Go directly to dropoff-code view which shows the code
   473→          setView("dropoff-code")
   474→          setPaymentResult(null) // Clear payment result
   475→          return
   476→        }
   477→
   478→        if (data.status === "failed") {
   479→          setError("Payment failed. Please try again.")
   480→          return
   481→        }
   482→
   483→        // Exponential backoff: 5s for first 10 attempts, then 8s, then 10s
   484→        const delay = attempts < 10 ? 5000 : attempts < 25 ? 8000 : 10000
   485→        pollingTimerRef.current = setTimeout(poll, delay)
   486→      } catch {
   487→        // Continue polling on error with longer delay
   488→        pollingTimerRef.current = setTimeout(poll, 8000)
   489→      }
   490→    }
   491→
   492→    poll()
   493→  }
