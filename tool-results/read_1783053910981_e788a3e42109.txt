   357→   357→                settings.setMixedContentMode(WebSettings.MIXED_CONTENT_ALWAYS_ALLOW);

   365→   365→
   366→   366→            try {

   373→   373→            webView.setWebChromeClient(new KioskWebChromeClient());

   385→   385→    private class KioskWebViewClient extends WebViewClient {
   386→   386→        @Override
   387→   387→        public void onPageFinished(WebView view, String url) {
   388→   388→            super.onPageFinished(view, url);
   389→   389→            if (url == null) return;
   390→   390→            if (OFFLINE_URL.equals(url)) {
   391→   391→                isShowingOfflinePage = true;
   392→   392→                Log.w(TAG, "Showing offline page");
   393→   393→            } else {
   394→   394→                isShowingOfflinePage = false;
   395→   395→                reconnectAttempts = 0;
   396→   396→                Log.i(TAG, "Page loaded: " + url);
   397→   397→            }
   398→   398→        }
   399→   399→
   400→   400→        @Override
   401→   401→        public void onReceivedError(WebView view, WebResourceRequest request, WebResourceError error) {
   402→   402→            if (request != null && request.isForMainFrame()) {
   403→   403→                if (!isShowingOfflinePage) showOfflinePage();
   404→   404→            }
   405→   405→        }
   406→   406→
   407→   407→        @Override
   408→   408→        public void onReceivedError(WebView view, int errorCode, String description, String failingUrl) {
   409→   409→            if (!isShowingOfflinePage) showOfflinePage();
   410→   410→        }
   411→   411→
   412→   412→        @Override
   413→   413→        public boolean shouldOverrideUrlLoading(WebView view, WebResourceRequest request) {
   414→   414→            if (request == null) return false;
   415→   415→            String url = request.getUrl().toString();
   416→   416→            if (url == null) return true;
   417→   417→            // Allow kiosk domain, backend, assets, and payment API
   418→   418→            if (url.startsWith("https://pickuplocker.vercel.app") ||
   419→   419→                url.startsWith("file:///android_asset/") ||
   420→   420→                url.startsWith("https://api.dimepay.app")) {
   421→   421→                return false;
   422→   422→            }
   423→   423→            Log.w(TAG, "Blocked navigation to: " + url);
   424→   424→            return true;
   425→   425→        }
   426→   426→    }

   446→   446→    // OFFLINE & RECONNECT
   447→   447→    // ============================================
   448→   448→
   449→   449→    private void showOfflinePage() {
   450→   450→        isShowingOfflinePage = true;
   451→   451→        if (webView != null && mainHandler != null) {
   452→   452→            mainHandler.post(() -> {
   453→   453→                if (webView != null && !isDestroyed) {
   454→   454→                    try { webView.loadUrl(OFFLINE_URL); } catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   455→   455→                }
   456→   456→            });
   457→   457→        }
   458→   458→        scheduleReconnect();
   459→   459→    }
   460→   460→
   461→   461→    private void scheduleReconnect() {
   462→   462→        if (reconnectAttempts >= MAX_RECONNECT_ATTEMPTS) return;
   463→   463→        reconnectAttempts++;
   464→   464→        if (mainHandler != null) {
   465→   465→            mainHandler.postDelayed(() -> {
   466→   466→                if (isShowingOfflinePage && !isDestroyed) checkAndReload();
   467→   467→            }, RECONNECT_DELAY_MS);
   468→   468→        }
   469→   469→    }
   470→   470→
   471→   471→    private void checkAndReload() {
   472→   472→        if (!isNetworkAvailable()) { scheduleReconnect(); return; }
   473→   473→        if (webView != null && mainHandler != null) {
   474→   474→            mainHandler.post(() -> {
   475→   475→                if (webView != null && !isDestroyed) {
   476→   476→                    try { webView.loadUrl(isViewingBackend ? getBackendUrl() : getKioskUrl()); }
   477→   477→                    catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   478→   478→                }
   479→   479→            });
   480→   480→        }
   481→   481→    }

   487→   487→    private void registerConnectivityMonitoring() {
   488→   488→        try {
   489→   489→            ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   490→   490→            if (cm == null) return;
   491→   491→            NetworkRequest request = new NetworkRequest.Builder()
   492→   492→                .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   493→   493→                .build();
   494→   494→            networkCallback = new ConnectivityManager.NetworkCallback() {
   495→   495→                @Override
   496→   496→                public void onAvailable(Network network) {
   497→   497→                    if (isShowingOfflinePage && mainHandler != null) {
   498→   498→                        mainHandler.postDelayed(() -> {
   499→   499→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   500→   500→                                reconnectAttempts = 0;
   501→   501→                                mainHandler.post(() -> {
   502→   502→                                    if (webView != null && !isDestroyed)
   503→   503→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   504→   504→                                });
   505→   505→                            }
   506→   506→                        }, CONNECTIVITY_CHECK_MS);
   507→   507→                    }
   508→   508→                }
   509→   509→                @Override
   510→   510→                public void onLost(Network network) { Log.w(TAG, "Network lost"); }
   511→   511→                @Override
   512→   512→                public void onCapabilitiesChanged(Network network, NetworkCapabilities caps) {
   513→   513→                    if (caps == null) return;
   514→   514→                    boolean ok = caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   515→   515→                            && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED);
   516→   516→                    if (ok && isShowingOfflinePage && mainHandler != null) {
   517→   517→                        mainHandler.postDelayed(() -> {
   518→   518→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   519→   519→                                reconnectAttempts = 0;
   520→   520→                                mainHandler.post(() -> {
   521→   521→                                    if (webView != null && !isDestroyed)
   522→   522→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   523→   523→                                });
   524→   524→                            }
   525→   525→                        }, CONNECTIVITY_CHECK_MS);
   526→   526→                    }
   527→   527→                }
   528→   528→            };
   529→   529→            cm.registerNetworkCallback(request, networkCallback);
   530→   530→        } catch (Throwable t) {
   531→   531→            Log.e(TAG, "Network monitoring failed: " + t.getMessage());
   532→   532→        }
   533→   533→    }
   534→   534→
   535→   535→    private void unregisterConnectivityMonitoring() {
   536→   536→        if (networkCallback != null) {
   537→   537→            try {
   538→   538→                ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   539→   539→                if (cm != null) cm.unregisterNetworkCallback(networkCallback);
   540→   540→            } catch (Throwable t) { Log.w(TAG, t.getMessage()); }
   541→   541→        }
   542→   542→    }

   576→   576→                float x = ev.getRawX();
   577→   577→                float y = ev.getRawY();
   578→   578→                if (x < hitAreaPx && y < hitAreaPx) {
   579→   579→                    long now = System.currentTimeMillis();
   580→   580→                    if (now - lastSecretTapTime > SECRET_TAP_TIMEOUT_MS) {
   581→   581→                        secretTapCount = 1;
   582→   582→                    } else {
   583→   583→                        secretTapCount++;
   584→   584→                    }
   585→   585→                    lastSecretTapTime = now;
   586→   586→                    if (secretTapCount >= SECRET_TAP_COUNT) {
   587→   587→                        secretTapCount = 0;
   588→   588→                        Log.i(TAG, "Secret trigger activated");
   589→   589→                        showPinDialog();
   590→   590→                    }
   591→   591→                }
   592→   592→            } catch (Throwable t) {
   593→   593→                Log.w(TAG, "Touch handler error: " + t.getMessage());
   594→   594→            }
   595→   595→        }
   596→   596→        return super.dispatchTouchEvent(ev);
   597→   597→    }

   663→   663→    private void handleWrongPin() {
   664→   664→        int attempts = getPrefs().getInt(KEY_PIN_FAILED_ATTEMPTS, 0) + 1;
   665→   665→        getPrefs().edit().putInt(KEY_PIN_FAILED_ATTEMPTS, attempts).apply();
   666→   666→        Log.w(TAG, "Wrong PIN attempt " + attempts + "/" + MAX_PIN_ATTEMPTS);
   667→   667→
   668→   668→        if (attempts >= MAX_PIN_ATTEMPTS) {
   669→   669→            // Lock out for 5 minutes
   670→   670→            long lockoutUntil = System.currentTimeMillis() + PIN_LOCKOUT_DURATION_MS;
   671→   671→            getPrefs().edit()
   672→   672→                .putLong(KEY_PIN_LOCKOUT_UNTIL, lockoutUntil)
   673→   673→                .putInt(KEY_PIN_FAILED_ATTEMPTS, 0)
   674→   674→                .apply();
   675→   675→            Toast.makeText(this, "Too many attempts. Locked for 5 minutes.", Toast.LENGTH_LONG).show();
   676→   676→        } else {
   677→   677→            int remaining = MAX_PIN_ATTEMPTS - attempts;
   678→   678→            new AlertDialog.Builder(this)
   679→   679→                .setTitle("Incorrect Code")
   680→   680→                .setMessage("The code you entered is incorrect.\n" + remaining + " attempt(s) remaining.")
   681→   681→                .setPositiveButton("OK", null)
   682→   682→                .show();
   683→   683→        }
   684→   684→    }
