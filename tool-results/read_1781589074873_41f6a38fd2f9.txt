    10→    10→// Fetch with timeout for Bestwond API calls
    11→    11→async function fetchWithTimeout(url: string, options: RequestInit): Promise<Response> {
    12→    12→  const controller = new AbortController();
    13→    13→  const timeoutId = setTimeout(() => controller.abort(), BESTWOND_TIMEOUT_MS);
    14→    14→  try {
    15→    15→    const response = await fetch(url, { ...options, signal: controller.signal });
    16→    16→    return response;
    17→    17→  } finally {
    18→    18→    clearTimeout(timeoutId);
    19→    19→  }
    20→    20→}

    41→    41→// Generate SHA512 signature for Bestwond API
    42→    42→// Signature = SHA512(sorted_url_encoded_params + app_secret)
    43→    43→export async function generateSignature(params: Record<string, string | number>, appSecret: string): Promise<string> {
    44→    44→  // Sort parameters alphabetically by key
    45→    45→  const sortedKeys = Object.keys(params).sort();
    46→    46→  
    47→    47→  // Build URL-encoded string
    48→    48→  const encodedParams = sortedKeys
    49→    49→    .map(key => `${encodeURIComponent(key)}=${encodeURIComponent(params[key])}`)
    50→    50→    .join('&');
    51→    51→  
    52→    52→  // Append app secret
    53→    53→  const stringToSign = `${encodedParams}${appSecret}`;
    54→    54→  
    55→    55→  // SHA512 hash
    56→    56→  const encoder = new TextEncoder();
    57→    57→  const dataBuffer = encoder.encode(stringToSign);
    58→    58→  const hashBuffer = await crypto.subtle.digest('SHA-512', dataBuffer);
    59→    59→  const hashArray = Array.from(new Uint8Array(hashBuffer));
    60→    60→  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
    61→    61→}

   135→   135→// Verify webhook signature from Bestwond
   136→   136→export function verifyWebhookSignature(payload: string, signature: string): boolean {
   137→   137→  const config = getConfig();
   138→   138→  
   139→   139→  if (!config.appSecret) {
   140→   140→    console.error('Bestwond app secret not configured - webhook verification failed');
   141→   141→    return false; // Reject unsigned webhooks in production
   142→   142→  }
   143→   143→  
   144→   144→  // Bestwond signs webhooks with SHA512(payload + appSecret)
   145→   145→  const expectedSignature = crypto
   146→   146→    .createHmac('sha512', config.appSecret)
   147→   147→    .update(payload)
   148→   148→    .digest('hex');
   149→   149→  
   150→   150→  return signature === expectedSignature;
   151→   151→}

   310→   310→// Open box and verify door actually opened
   311→   311→export async function openBoxAndVerify(
   312→   312→  deviceNumber: string,
   313→   313→  boxNo: number,
   314→   314→  credentials: BestwondCredentials,
   315→   315→  maxRetries: number = 3
   316→   316→): Promise<{ success: boolean; message: string; doorStatus?: string; apiResponse?: BestwondResponse }> {
   317→   317→  // First, check if device is online
   318→   318→  const statusResult = await getDeviceStatusWithCredentials(deviceNumber, credentials);
   319→   319→  if (statusResult.code !== 0) {
   320→   320→    return { 
   321→   321→      success: false, 
   322→   322→      message: `Cannot reach device. API error: ${statusResult.msg || 'Unknown error'}` 
   323→   323→    };
   324→   324→  }
   325→   325→  
   326→   326→  const deviceData = statusResult.data;
   327→   327→  const isOnline = deviceData?.status === 'on' || deviceData?.online === true;
   328→   328→  
   329→   329→  if (!isOnline) {
   330→   330→    return { 
   331→   331→      success: false, 
   332→   332→      message: `Device ${deviceNumber} is OFFLINE. Please check power and network connection.` 
   333→   333→    };
   334→   334→  }
   335→   335→  
   336→   336→  // Get the lock_address for this box
   337→   337→  const boxListResult = await getBoxListWithCredentials(deviceNumber, credentials);
   338→   338→  let lockAddress = `01${boxNo.toString(16).toLowerCase().padStart(2, '0')}`; // HEX format
   339→   339→  
   340→   340→  if (boxListResult.code === 0 && boxListResult.data) {
   341→   341→    const box = boxListResult.data.find(b => parseInt(b.box_name, 10) === boxNo);
   342→   342→    if (box?.lock_address) {
   343→   343→      lockAddress = box.lock_address;
   344→   344→    }
   345→   345→  }
   346→   346→  
   347→   347→  // Try to open the box
   348→   348→  let lastError: string | undefined;
   349→   349→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
   350→   350→    console.log(`Open box attempt ${attempt}/${maxRetries}`);
   351→   351→    
   352→   352→    const openResult = await openBoxWithCredentials(deviceNumber, boxNo, credentials);
   353→   353→    
   354→   354→    // Check both API response code AND device status
   355→   355→    const resultData = openResult.data as { status?: string; msg?: string } | undefined;
   356→   356→    const deviceStatus = resultData?.status;
   357→   357→    const deviceMsg = resultData?.msg;
   358→   358→    
   359→   359→    if (openResult.code === 0 && deviceStatus === 'success') {
   360→   360→      // API and device both say success - wait and check door
   361→   361→      await new Promise(resolve => setTimeout(resolve, 1500));
   362→   362→      
   363→   363→      const doorStatus = await getDoorStatusWithCredentials(deviceNumber, lockAddress, credentials);
   364→   364→      
   365→   365→      return {
   366→   366→        success: true,
   367→   367→        message: `Box #${boxNo} opened successfully.`,
   368→   368→        doorStatus: doorStatus.data?.status,
   369→   369→        apiResponse: openResult
   370→   370→      };
   371→   371→    }
   372→   372→    
   373→   373→    // Check for specific device errors
   374→   374→    if (deviceStatus === 'fail') {
   375→   375→      // Device rejected the command - no point retrying
   376→   376→      if (deviceMsg?.includes('uqkey') || deviceMsg?.includes('key')) {
   377→   377→        return {
   378→   378→          success: false,
   379→   379→          message: `❌ DEVICE NOT LINKED: This device is not registered to your app account. Contact Bestwond support to link device ${deviceNumber} to app ${credentials.appId}. Error: ${deviceMsg}`,
   380→   380→          apiResponse: openResult
   381→   381→        };
   382→   382→      }
   383→   383→      
   384→   384→      if (deviceMsg?.includes('offline') || deviceMsg?.includes('not online')) {
   385→   385→        return {
   386→   386→          success: false,
   387→   387→          message: `❌ DEVICE OFFLINE: Device ${deviceNumber} is not connected. Check power and network.`,
   388→   388→          apiResponse: openResult
   389→   389→        };
   390→   390→      }
   391→   391→      
   392→   392→      // Other device error
   393→   393→      lastError = deviceMsg || 'Device rejected command';
   394→   394→    } else {
   395→   395→      lastError = openResult.msg || `Error code: ${openResult.code}`;
   396→   396→    }
   397→   397→    
   398→   398→    console.log(`Attempt ${attempt} failed:`, lastError);
   399→   399→    
   400→   400→    // Wait before retry
   401→   401→    if (attempt < maxRetries) {
   402→   402→      await new Promise(resolve => setTimeout(resolve, 1000));
   403→   403→    }
   404→   404→  }
   405→   405→  
   406→   406→  return {
   407→   407→    success: false,
   408→   408→    message: `Failed to open box after ${maxRetries} attempts. Last error: ${lastError}`
   409→   409→  };
   410→   410→}
