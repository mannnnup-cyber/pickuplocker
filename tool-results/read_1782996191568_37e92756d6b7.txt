   556→   556→            window.location.href = resp.redirectUrl || '/kiosk-lite?action=payment-success&msg=' + encodeURIComponent(resp.message || 'Payment confirmed!');

   560→   560→          }
   561→   561→        } catch(e) {
   562→   562→          // Non-JSON response — ignore, keep polling
   563→   563→        }
   564→   564→      };
