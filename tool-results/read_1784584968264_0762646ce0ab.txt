   225→    private void applySavedOrientation() {
   226→        String orient = getOrientation();
   227→        switch (orient) {
   228→            case "portrait":
   229→                setRequestedOrientation(android.content.pm.ActivityInfo.SCREEN_ORIENTATION_SENSOR_PORTRAIT);
   230→                break;
   231→            case "landscape":
   232→                setRequestedOrientation(android.content.pm.ActivityInfo.SCREEN_ORIENTATION_SENSOR_LANDSCAPE);
   233→                break;
   234→            case "auto":
   235→                setRequestedOrientation(android.content.pm.ActivityInfo.SCREEN_ORIENTATION_FULL_SENSOR);
   236→                break;
   237→            default:
   238→                setRequestedOrientation(android.content.pm.ActivityInfo.SCREEN_ORIENTATION_SENSOR_PORTRAIT);
   239→                break;
   240→        }
   241→    }

   356→    private void configureWebView() {
   357→        if (webView == null) return;
   358→        try {
   359→            WebSettings settings = webView.getSettings();
   360→            settings.setJavaScriptEnabled(true);
   361→            settings.setDomStorageEnabled(true);
   362→            settings.setAllowFileAccess(true);
   363→            settings.setAllowContentAccess(true);
   364→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
   365→                settings.setMixedContentMode(WebSettings.MIXED_CONTENT_ALWAYS_ALLOW);
   366→            }
   367→            settings.setCacheMode(WebSettings.LOAD_DEFAULT);
   368→            settings.setBuiltInZoomControls(false);
   369→            settings.setDisplayZoomControls(false);
   370→
   371→            String userAgent = settings.getUserAgentString();
   372→            settings.setUserAgentString(userAgent + " PickupKiosk/" + APP_VERSION);
   373→
   374→            try {
   375→                boolean isDebug = false;
   376→                try { isDebug = BuildConfig.DEBUG; } catch (Throwable ignored) {}
   377→                WebView.setWebContentsDebuggingEnabled(isDebug);
   378→            } catch (Throwable t) { Log.w(TAG, "WebView debugging: " + t.getMessage()); }
   379→
   380→            webView.setWebViewClient(new KioskWebViewClient());
   381→            webView.setWebChromeClient(new KioskWebChromeClient());
   382→            webView.loadUrl(getKioskUrl());
   383→        } catch (Throwable t) {
   384→            Log.e(TAG, "WebView config failed: " + t.getMessage(), t);
   385→            showErrorScreen("WebView configuration failed: " + t.getMessage());
   386→        }
   387→    }

   393→    private class KioskWebViewClient extends WebViewClient {
   394→        @Override
   395→        public void onPageFinished(WebView view, String url) {
   396→            super.onPageFinished(view, url);
   397→            if (url == null) return;
   398→            if (OFFLINE_URL.equals(url)) {
   399→                isShowingOfflinePage = true;
   400→                Log.w(TAG, "Showing offline page");
   401→            } else {
   402→                isShowingOfflinePage = false;
   403→                reconnectAttempts = 0;
   404→                Log.i(TAG, "Page loaded: " + url);
   405→            }
   406→        }
   407→
   408→        @Override
   409→        public void onReceivedError(WebView view, WebResourceRequest request, WebResourceError error) {
   410→            if (request != null && request.isForMainFrame()) {
   411→                if (!isShowingOfflinePage) showOfflinePage();
   412→            }
   413→        }
   414→
   415→        @Override
   416→        public void onReceivedError(WebView view, int errorCode, String description, String failingUrl) {
   417→            if (!isShowingOfflinePage) showOfflinePage();
   418→        }
   419→
   420→        @Override
   421→        public boolean shouldOverrideUrlLoading(WebView view, WebResourceRequest request) {
   422→            if (request == null) return false;
   423→            String url = request.getUrl().toString();
   424→            if (url == null) return true;
   425→            // Allow kiosk domain, backend, assets, and payment API
   426→            if (url.startsWith("https://pickuplocker.vercel.app") ||
   427→                url.startsWith("file:///android_asset/") ||
   428→                url.startsWith("https://api.dimepay.app")) {
   429→                return false;
   430→            }
   431→            Log.w(TAG, "Blocked navigation to: " + url);
   432→            return true;
   433→        }
   434→    }

   454→    // OFFLINE & RECONNECT
   455→    // ============================================
   456→
   457→    private void showOfflinePage() {
   458→        isShowingOfflinePage = true;
   459→        if (webView != null && mainHandler != null) {
   460→            mainHandler.post(() -> {
   461→                if (webView != null && !isDestroyed) {
   462→                    try { webView.loadUrl(OFFLINE_URL); } catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   463→                }
   464→            });
   465→        }
   466→        scheduleReconnect();
   467→    }
   468→
   469→    private void scheduleReconnect() {
   470→        if (reconnectAttempts >= MAX_RECONNECT_ATTEMPTS) return;
   471→        reconnectAttempts++;
   472→        if (mainHandler != null) {
   473→            mainHandler.postDelayed(() -> {
   474→                if (isShowingOfflinePage && !isDestroyed) checkAndReload();
   475→            }, RECONNECT_DELAY_MS);
   476→        }
   477→    }
   478→
   479→    private void checkAndReload() {
   480→        if (!isNetworkAvailable()) { scheduleReconnect(); return; }
   481→        if (webView != null && mainHandler != null) {
   482→            mainHandler.post(() -> {
   483→                if (webView != null && !isDestroyed) {
   484→                    try { webView.loadUrl(isViewingBackend ? getBackendUrl() : getKioskUrl()); }
   485→                    catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   486→                }
   487→            });
   488→        }
   489→    }

   491→    // ============================================
   492→    // CONNECTIVITY MONITORING
   493→    // ============================================
   494→
   495→    private void registerConnectivityMonitoring() {
   496→        try {
   497→            ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   498→            if (cm == null) return;
   499→            NetworkRequest request = new NetworkRequest.Builder()
   500→                .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   501→                .build();
   502→            networkCallback = new ConnectivityManager.NetworkCallback() {
   503→                @Override
   504→                public void onAvailable(Network network) {
   505→                    if (isShowingOfflinePage && mainHandler != null) {
   506→                        mainHandler.postDelayed(() -> {
   507→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   508→                                reconnectAttempts = 0;
   509→                                mainHandler.post(() -> {
   510→                                    if (webView != null && !isDestroyed)
   511→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   512→                                });
   513→                            }
   514→                        }, CONNECTIVITY_CHECK_MS);
   515→                    }
   516→                }
   517→                @Override
   518→                public void onLost(Network network) { Log.w(TAG, "Network lost"); }
   519→                @Override
   520→                public void onCapabilitiesChanged(Network network, NetworkCapabilities caps) {
   521→                    if (caps == null) return;
   522→                    boolean ok = caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   523→                            && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED);
   524→                    if (ok && isShowingOfflinePage && mainHandler != null) {
   525→                        mainHandler.postDelayed(() -> {
   526→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   527→                                reconnectAttempts = 0;
   528→                                mainHandler.post(() -> {
   529→                                    if (webView != null && !isDestroyed)
   530→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   531→                                });
   532→                            }
   533→                        }, CONNECTIVITY_CHECK_MS);
   534→                    }
   535→                }
   536→            };
   537→            cm.registerNetworkCallback(request, networkCallback);
   538→        } catch (Throwable t) {
   539→            Log.e(TAG, "Network monitoring failed: " + t.getMessage());
   540→        }
   541→    }
   542→
   543→    private void unregisterConnectivityMonitoring() {
   544→        if (networkCallback != null) {
   545→            try {
   546→                ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   547→                if (cm != null) cm.unregisterNetworkCallback(networkCallback);
   548→            } catch (Throwable t) { Log.w(TAG, t.getMessage()); }
   549→        }
   550→    }
   551→
   552→    private boolean isNetworkAvailable() {
   553→        try {
   554→            ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   555→            if (cm == null) return false;
   556→            Network activeNetwork = cm.getActiveNetwork();
   557→            if (activeNetwork == null) return false;
   558→            NetworkCapabilities caps = cm.getNetworkCapabilities(activeNetwork);
   559→            return caps != null && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET);
   560→        } catch (Throwable t) { return true; }
   561→    }

   564→    // WEBVIEW HEALTH MONITORING — Detect frozen/white screens
   565→    // ============================================
   566→
   567→    /**
   568→     * Starts periodic health checks on the WebView.
   569→     * If the WebView appears unresponsive or has blank content for
   570→     * multiple consecutive checks, force a reload.
   571→     */
   572→    private void startHealthMonitoring() {
   573→        healthCheckRunnable = new Runnable() {
   574→            @Override
   575→            public void run() {
   576→                if (!isDestroyed) {
   577→                    checkWebViewHealth();
   578→                    mainHandler.postDelayed(this, HEALTH_CHECK_INTERVAL_MS);
   579→                }
   580→            }
   581→        };
   582→        mainHandler.postDelayed(healthCheckRunnable, HEALTH_CHECK_INTERVAL_MS);
   583→        Log.i(TAG, "WebView health monitoring started (interval: " + HEALTH_CHECK_INTERVAL_MS + "ms)");
   584→    }
   585→
   586→    private void stopHealthMonitoring() {
   587→        if (healthCheckRunnable != null) {
   588→            mainHandler.removeCallbacks(healthCheckRunnable);
   589→            healthCheckRunnable = null;
   590→            Log.i(TAG, "WebView health monitoring stopped");
   591→        }
   592→    }

   594→    /**
   595→     * Check if the WebView is healthy by evaluating its content.
   596→     * If the page appears blank or frozen for several consecutive checks,
   597→     * force a reload to recover from white screen issues.
   598→     */
   599→    private void checkWebViewHealth() {
   600→        if (webView == null) {
   601→            healthCheckFailCount++;
   602→            Log.w(TAG, "Health check: WebView is null (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   603→            if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   604→                Log.w(TAG, "Health check: Too many failures — attempting full recovery");
   605→                healthCheckFailCount = 0;
   606→                reloadKiosk();
   607→            }
   608→            return;
   609→        }
   610→
   611→        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.KITKAT) {
   612→            webView.evaluateJavascript("(function(){ try { var c = document.querySelector('.container'); return c ? 'ok' : 'empty'; } catch(e) { return 'error'; } })()", new android.webkit.ValueCallback<String>() {
   613→                @Override
   614→                public void onReceiveValue(String value) {
   615→                    if (value != null && (value.contains("ok") || value.contains("empty"))) {
   616→                        if (value.contains("empty")) {
   617→                            healthCheckFailCount++;
   618→                            Log.w(TAG, "Health check: Page content empty (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   619→                        } else {
   620→                            if (healthCheckFailCount > 0) {
   621→                                Log.i(TAG, "Health check: Page recovered (was " + healthCheckFailCount + " fails)");
   622→                            }
   623→                            healthCheckFailCount = 0;
   624→                        }
   625→                    } else {
   626→                        healthCheckFailCount++;
   627→                        Log.w(TAG, "Health check: No response or unexpected value: " + value + " (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   628→                    }
   629→
   630→                    if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   631→                        Log.w(TAG, "Health check: Threshold reached — force reloading kiosk");
   632→                        healthCheckFailCount = 0;
   633→                        reloadKiosk();
   634→                    }
   635→                }
   636→            });
   637→        } else {
   638→            // Pre-KitKat: just check offline status
   639→            if (isShowingOfflinePage) {
   640→                healthCheckFailCount++;
   641→                if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   642→                    healthCheckFailCount = 0;
   643→                    reloadKiosk();
   644→                }
   645→            } else {
   646→                healthCheckFailCount = 0;
   647→            }
   648→        }
   649→    }

   666→    @Override
   667→    public boolean dispatchTouchEvent(MotionEvent ev) {
   668→        if (ev.getAction() == MotionEvent.ACTION_DOWN) {
   669→            try {
   670→                float density = getResources().getDisplayMetrics().density;
   671→                float hitAreaPx = SECRET_TAP_AREA_DP * density;
   672→                float x = ev.getRawX();
   673→                float y = ev.getRawY();
   674→                if (x < hitAreaPx && y < hitAreaPx) {
   675→                    long now = System.currentTimeMillis();
   676→                    if (now - lastSecretTapTime > SECRET_TAP_TIMEOUT_MS) {
   677→                        secretTapCount = 1;
   678→                    } else {
   679→                        secretTapCount++;
   680→                    }
   681→                    lastSecretTapTime = now;
   682→                    if (secretTapCount >= SECRET_TAP_COUNT) {
   683→                        secretTapCount = 0;
   684→                        Log.i(TAG, "Secret trigger activated");
   685→                        showPinDialog();
   686→                    }
   687→                }
   688→            } catch (Throwable t) {
   689→                Log.w(TAG, "Touch handler error: " + t.getMessage());
   690→            }
   691→        }
   692→        return super.dispatchTouchEvent(ev);
   693→    }

   759→    private void handleWrongPin() {
   760→        int attempts = getPrefs().getInt(KEY_PIN_FAILED_ATTEMPTS, 0) + 1;
   761→        getPrefs().edit().putInt(KEY_PIN_FAILED_ATTEMPTS, attempts).apply();
   762→        Log.w(TAG, "Wrong PIN attempt " + attempts + "/" + MAX_PIN_ATTEMPTS);
   763→
   764→        if (attempts >= MAX_PIN_ATTEMPTS) {
   765→            // Lock out for 5 minutes
   766→            long lockoutUntil = System.currentTimeMillis() + PIN_LOCKOUT_DURATION_MS;
   767→            getPrefs().edit()
   768→                .putLong(KEY_PIN_LOCKOUT_UNTIL, lockoutUntil)
   769→                .putInt(KEY_PIN_FAILED_ATTEMPTS, 0)
   770→                .apply();
   771→            Toast.makeText(this, "Too many attempts. Locked for 5 minutes.", Toast.LENGTH_LONG).show();
   772→        } else {
   773→            int remaining = MAX_PIN_ATTEMPTS - attempts;
   774→            new AlertDialog.Builder(this)
   775→                .setTitle("Incorrect Code")
   776→                .setMessage("The code you entered is incorrect.\n" + remaining + " attempt(s) remaining.")
   777→                .setPositiveButton("OK", null)
   778→                .show();
   779→        }
   780→    }
