    35→  if (!str) return '';

    39→    .replace(/>/g, '&gt;')
    40→    .replace(/"/g, '&quot;')

   244→// Kiosk-mode JavaScript - Android 5.0 compatible (ES5 only, no arrow functions, no const/let, no Promises)
   245→const KIOSK_JS = `
   246→  var inactivityTimer;
   247→  var INACTIVITY_MS = 90000;
   248→  function resetInactivityTimer() {
   249→    clearTimeout(inactivityTimer);
   250→    inactivityTimer = setTimeout(function() {
   251→      window.location.href = '/kiosk-lite';
   252→    }, INACTIVITY_MS);
   253→  }
   254→  document.addEventListener('touchstart', resetInactivityTimer, false);
   255→  document.addEventListener('click', resetInactivityTimer, false);
   256→  document.addEventListener('keydown', resetInactivityTimer, false);
   257→  resetInactivityTimer();
   258→  function updateClock() {
   259→    var el = document.getElementById('kiosk-clock');
   260→    if (!el) return;
   261→    var now = new Date();
   262→    var h = now.getHours();
   263→    var m = now.getMinutes();
   264→    var ampm = h >= 12 ? 'PM' : 'AM';
   265→    h = h % 12;
   266→    if (h === 0) h = 12;
   267→    if (m < 10) m = '0' + m;
   268→    el.textContent = h + ':' + m + ' ' + ampm;
   269→  }
   270→  updateClock();
   271→  setInterval(updateClock, 30000);
   272→
   273→  // ---- AJAX Payment Polling (Chrome 37 compatible) ----
   274→  // Polls /api/kiosk/payment-status every N seconds.
   275→  // When payment is completed, redirects to the appropriate screen.
   276→  // The QR code stays visible on the page while polling runs.
   277→  function startPaymentPolling(pollUrl, checkIntervalMs) {
   278→    var pollInterval = setInterval(function() {
   279→      var xhr = new XMLHttpRequest();
   280→      xhr.open('GET', pollUrl, true);
   281→      xhr.setRequestHeader('Accept', 'application/json');
   282→      xhr.onreadystatechange = function() {
   283→        if (xhr.readyState !== 4) return;
   284→        if (xhr.status !== 200) return;
   285→        try {
   286→          var resp = JSON.parse(xhr.responseText);
   287→          if (resp && resp.status === 'completed') {
   288→            clearInterval(pollInterval);
   289→            // Stop the inactivity timer while redirecting
   290→            clearTimeout(inactivityTimer);
   291→            // Redirect to the appropriate next screen
   292→            window.location.href = resp.redirectUrl || '/kiosk-lite?action=payment-success&msg=' + encodeURIComponent(resp.message || 'Payment confirmed!');
   293→          } else if (resp && resp.status === 'failed') {
   294→            clearInterval(pollInterval);
   295→            window.location.href = '/kiosk-lite?action=error&msg=' + encodeURIComponent(resp.message || 'Payment failed');
   296→          } else if (resp && resp.status === 'not_found') {
   297→            clearInterval(pollInterval);
   298→            window.location.href = '/kiosk-lite?action=error&msg=' + encodeURIComponent('Payment session not found');
   299→          }
   300→          // status === 'pending' → keep polling, do nothing
   301→        } catch(e) {
   302→          // Non-JSON response — ignore, keep polling
   303→        }
   304→      };
   305→      xhr.send();
   306→    }, checkIntervalMs || 5000);
   307→  }
   308→`;

   359→export async function GET(request: NextRequest) {
   360→  const { searchParams } = request.nextUrl;
   361→  const flow = searchParams.get('flow');
   362→  const step = searchParams.get('step');
   363→
   364→  // Check drop-off payment status (polling via meta refresh)
   365→  if (flow === 'dropoff' && step === 'check_payment') {
   366→    return await handleCheckPayment(searchParams.get('paymentId') || '');
   367→  }
   368→
   369→  // Check storage fee payment status (polling via meta refresh)
   370→  if (flow === 'pickup' && step === 'check_storage_payment') {
   371→    return await handleCheckStoragePayment(searchParams.get('paymentId') || '');
   372→  }
   373→
   374→  // Default – redirect to kiosk home
   375→  return NextResponse.redirect(new URL('/kiosk-lite', request.url), 303);
   376→}

   378→// ============================================
   379→// POST handler – all form submissions
   380→// ============================================
   381→export async function POST(request: NextRequest) {
   382→  const formData = await request.formData();
   383→  const flow = formData.get('flow') as string;
   384→  const step = formData.get('step') as string;
   385→
   386→  try {
   387→    // ============================================
   388→    // DROP-OFF FLOW
   389→    // ============================================
   390→    if (flow === 'dropoff') {
   391→      return await handleDropoffFlow(step, formData, request);
   392→    }
   393→
   394→    // ============================================
   395→    // PICKUP FLOW
   396→    // ============================================
   397→    if (flow === 'pickup') {
   398→      return await handlePickupFlow(step, formData, request);
   399→    }
   400→  } catch (error) {
   401→    console.error('Kiosk action error:', error);
   402→    return htmlResponse(`
   403→      <h2 class="title">Error</h2>
   404→      <p class="error-msg">Something went wrong. Please try again.</p>
   405→      <a href="/kiosk-lite" class="btn btn-primary">Back to Home</a>
   406→    `);
   407→  }
   408→
   409→  // Fallback
   410→  return NextResponse.redirect(new URL('/kiosk-lite', request.url), 303);
   411→}

   413→// ============================================
   414→// DROP-OFF FLOW HANDLER
   415→// ============================================
   416→async function handleDropoffFlow(
   417→  step: string,
   418→  formData: FormData,
   419→  request: NextRequest
   420→): Promise<NextResponse> {
   421→  // step=1 → Redirect to drop-off page (303 = See Other, browser follows with GET)
   422→  if (step === '1') {
   423→    return NextResponse.redirect(
   424→      new URL('/kiosk-lite?action=dropoff', request.url), 303
   425→    );
   426→  }
   427→
   428→  // step=enter_save_code → Redirect to save-code entry page
   429→  if (step === 'enter_save_code') {
   430→    return NextResponse.redirect(
   431→      new URL('/kiosk-lite?action=dropoff-save', request.url), 303
   432→    );
   433→  }
   434→
   435→  // step=select_size → Redirect to box size selection page
   436→  if (step === 'select_size') {
   437→    return NextResponse.redirect(
   438→      new URL('/kiosk-lite?action=dropoff-size', request.url), 303
   439→    );
   440→  }
   441→
   442→  // step=buy_enter_phone → Redirect to phone entry page with boxSize
   443→  if (step === 'buy_enter_phone') {
   444→    const boxSize = formData.get('boxSize') as string;
   445→    return NextResponse.redirect(
   446→      new URL(`/kiosk-lite?action=dropoff-phone&boxSize=${esc(boxSize)}`, request.url), 303
   447→    );
   448→  }
   449→
   450→  // step=courier_login → Redirect to courier login page
   451→  if (step === 'courier_login') {
   452→    return NextResponse.redirect(
   453→      new URL('/kiosk-lite?action=courier-login', request.url), 303
   454→    );
   455→  }
   456→
   457→  // step=use_save_code → Process save code drop-off
   458→  if (step === 'use_save_code') {
   459→    return await handleUseSaveCode(formData);
   460→  }
   461→
   462→  // step=buy_create_payment → Create DimePay payment, show QR
   463→  if (step === 'buy_create_payment') {
   464→    return await handleBuyCreatePayment(formData);
   465→  }
   466→
   467→  // step=check_payment → Check payment status
   468→  if (step === 'check_payment') {
   469→    const paymentId = formData.get('paymentId') as string;
   470→    return await handleCheckPayment(paymentId);
   471→  }
   472→
   473→  // step=open_after_payment → Open box after confirmed payment
   474→  if (step === 'open_after_payment') {
   475→    return await handleOpenAfterPayment(formData);
   476→  }
   477→
   478→  // step=courier_auth → Verify courier PIN
   479→  if (step === 'courier_auth') {
   480→    return await handleCourierAuth(formData);
   481→  }
   482→
   483→  // step=courier_set_pin → Set permanent PIN after temp PIN login
   484→  if (step === 'courier_set_pin') {
   485→    return await handleCourierSetPin(formData);
   486→  }
   487→
   488→  // step=courier_dropoff → Process courier drop-off
   489→  if (step === 'courier_dropoff') {
   490→    return await handleCourierDropoff(formData);
   491→  }
   492→
   493→  // Unknown step – redirect home
   494→  return NextResponse.redirect(new URL('/kiosk-lite', request.url), 303);
   495→}

   497→// ============================================
   498→// PICKUP FLOW HANDLER
   499→// ============================================
   500→async function handlePickupFlow(
   501→  step: string,
   502→  formData: FormData,
   503→  request: NextRequest
   504→): Promise<NextResponse> {
   505→  // step=1 → Redirect to pickup code entry page
   506→  if (step === '1') {
   507→    return NextResponse.redirect(
   508→      new URL('/kiosk-lite?action=pickup', request.url), 303
   509→    );
   510→  }
   511→
   512→  // step=lookup → Find order by pickCode
   513→  if (step === 'lookup') {
   514→    return await handlePickupLookup(formData);
   515→  }
   516→
   517→  // step=pay_storage_fee → Create storage fee payment, show QR
   518→  if (step === 'pay_storage_fee') {
   519→    return await handlePayStorageFee(formData);
   520→  }
   521→
   522→  // step=pay_storage_fee_qr → Pay storage fee via QR (no saved card)
   523→  if (step === 'pay_storage_fee_qr') {
   524→    return await handlePayStorageFeeQR(formData, request);
   525→  }
   526→
   527→  // step=charge_saved_card → Charge user's saved card token
   528→  if (step === 'charge_saved_card') {
   529→    return await handleChargeSavedCard(formData);
   530→  }
   531→
   532→  // step=check_storage_payment → Check storage fee payment status
   533→  if (step === 'check_storage_payment') {
   534→    const paymentId = formData.get('paymentId') as string;
   535→    return await handleCheckStoragePayment(paymentId);
   536→  }
   537→
   538→  // step=open_pickup → Open locker for pickup
   539→  if (step === 'open_pickup') {
   540→    return await handleOpenPickup(formData);
   541→  }
   542→
   543→  // Unknown step – redirect home
   544→  return NextResponse.redirect(new URL('/kiosk-lite', request.url), 303);
   545→}

  1324→// Hash PIN for verification — must match the hashing in /api/courier/pin and /api/courier/login
  1325→function hashPin(pin: string, phone: string): string {
  1326→  const secret = process.env.NEXTAUTH_SECRET
  1327→  if (!secret) {
  1328→    throw new Error("NEXTAUTH_SECRET environment variable is required")
  1329→  }
  1330→  return crypto
  1331→    .createHmac('sha256', secret)
  1332→    .update(`${phone}:${pin}`)
  1333→    .digest('hex');
  1334→}
