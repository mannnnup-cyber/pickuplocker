    94→    94→async function kioskFetch(url: string, options?: RequestInit, timeoutMs: number = FETCH_TIMEOUT_MS): Promise<Response> {
    95→    95→  const controller = new AbortController()
    96→    96→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
    97→    97→  try {
    98→    98→    const res = await fetch(url, { ...options, signal: controller.signal })
    99→    99→    if (!res.ok) {
   100→   100→      throw new Error(`HTTP ${res.status} from ${url}`)
   101→   101→    }
   102→   102→    return res
   103→   103→  } catch (error) {
   104→   104→    if (error instanceof DOMException && error.name === 'AbortError') {
   105→   105→      throw new Error(`Request to ${url} timed out after ${timeoutMs}ms`)
   106→   106→    }
   107→   107→    throw error
   108→   108→  } finally {
   109→   109→    clearTimeout(timeoutId)
   110→   110→  }
   111→   111→}

   215→   215→      try { localStorage.setItem(CRASH_KEY, '1') } catch {}
   216→   216→    }

   243→   243→      } else if (document.visibilityState === 'visible') {
   244→   244→        // Page is visible again
   245→   245→        if (hiddenAt) {
   246→   246→          const hiddenDuration = Date.now() - hiddenAt
   247→   247→          console.log('[Kiosk] Page visible again after', Math.round(hiddenDuration / 1000), 'seconds')
   248→   248→
   249→   249→          // If hidden for too long, do a full page reload to clear any corrupted state
   250→   250→          if (hiddenDuration > MAX_BACKGROUND_MS) {
   251→   251→            console.log('[Kiosk] Page was hidden too long. Reloading to prevent white screen...')
   252→   252→            window.location.reload()
   253→   253→            return
   254→   254→          }
   255→   255→
   256→   256→          // If hidden for a moderate time, refresh box availability and reset if on home screen

   268→   268→              }
   269→   269→            }
   270→   270→            refreshBoxes()
   271→   271→          }
   272→   272→        }
   273→   273→        hiddenAt = null
   274→   274→      }
   275→   275→    }
   276→   276→
   277→   277→    document.addEventListener('visibilitychange', handleVisibilityChange)
   278→   278→    return () => document.removeEventListener('visibilitychange', handleVisibilityChange)
   279→   279→  }, [])
   280→   280→
   281→   281→  // Periodic self-healing check — detects and recovers from white/frozen screen states
   282→   282→  useEffect(() => {
   283→   283→    const CHECK_INTERVAL = 2 * 60 * 1000 // Every 2 minutes
   284→   284→    const WHITE_SCREEN_CHECK_KEY = 'pickup_kiosk_alive'
   285→   285→
   286→   286→    const selfHeal = () => {
   287→   287→      try {
   288→   288→        const now = Date.now()
   289→   289→        localStorage.setItem(WHITE_SCREEN_CHECK_KEY, now.toString())
   290→   290→
   291→   291→        // Check if the DOM is actually rendering — if the root element has no children
   292→   292→        // or has collapsed, the page is likely frozen/white
   293→   293→        const rootEl = document.getElementById('__next') || document.getElementById('root')
   294→   294→        if (rootEl) {
   295→   295→          const hasContent = rootEl.children.length > 0 || rootEl.textContent?.trim()
   296→   296→          if (!hasContent) {
   297→   297→            console.error('[Kiosk] White screen detected — root element is empty. Forcing reload.')
   298→   298→            window.location.reload()
   299→   299→            return
   300→   300→          }
   301→   301→        }
   302→   302→
   303→   303→        // Check if we've been on the same non-home view for too long (possible frozen state)
   304→   304→        // This is a soft check — the auto-timeout should handle most cases,
   305→   305→        // but if timers are frozen, we need this fallback
   306→   306→        const lastHeartbeat = parseInt(localStorage.getItem('pickup_kiosk_heartbeat') || '0')
   307→   307→        if (lastHeartbeat && (now - lastHeartbeat > 10 * 60 * 1000)) {
   308→   308→          // No heartbeat update for 10 minutes means timers are likely frozen
   309→   309→          console.error('[Kiosk] No heartbeat for 10 minutes — timers appear frozen. Reloading.')
   310→   310→          window.location.reload()
   311→   311→        }
   312→   312→      } catch (err) {
   313→   313→        console.error('[Kiosk] Self-heal check failed:', err)
   314→   314→      }
   315→   315→    }
   316→   316→
   317→   317→    const interval = setInterval(selfHeal, CHECK_INTERVAL)
   318→   318→    // Run once immediately on mount
   319→   319→    selfHeal()
   320→   320→    return () => clearInterval(interval)
   321→   321→  }, [])
   322→   322→
   323→   323→  // Cleanup all polling timers on unmount
   324→   324→  useEffect(() => {
   325→   325→    isMountedRef.current = true
   326→   326→    return () => {
   327→   327→      isMountedRef.current = false
   328→   328→      // Clear any active polling timer
   329→   329→      if (pollingTimerRef.current) {
   330→   330→        clearTimeout(pollingTimerRef.current)
   331→   331→        pollingTimerRef.current = null
   332→   332→      }
   333→   333→      // Remove any dangling visibilitychange listener
   334→   334→      if (visibilityHandlerRef.current) {
   335→   335→        document.removeEventListener('visibilitychange', visibilityHandlerRef.current)
   336→   336→        visibilityHandlerRef.current = null
   337→   337→      }
   338→   338→    }
   339→   339→  }, [])
   340→   340→
   341→   341→  // Global unhandled rejection handler — prevents white screen from uncaught promise errors

   344→   344→      console.error('[Kiosk] Unhandled promise rejection:', event.reason)

   349→   349→      setLoadingBoxes(false)
   350→   350→      // Auto-recover after 5 seconds

   358→   358→
   359→   359→  // Stuck-state watchdog — detects when loading has been stuck too long and forces recovery

   362→   362→    const CHECK_INTERVAL_MS = 5000   // Check every 5 seconds
   363→   363→

   365→   365→      const isLoading = loading || loadingBoxes
   366→   366→      if (isLoading) {
   367→   367→        if (!loadingStartTimeRef.current) {
   368→   368→          loadingStartTimeRef.current = Date.now()
   369→   369→        }
   370→   370→        const stuckDuration = Date.now() - loadingStartTimeRef.current
   371→   371→        if (stuckDuration > STUCK_THRESHOLD_MS) {
   372→   372→          console.error(`[Kiosk] Stuck loading state detected for ${stuckDuration}ms. Forcing recovery.`)
   373→   373→          loadingStartTimeRef.current = null
   374→   374→          setLoading(false)
   375→   375→          setLoadingBoxes(false)
   376→   376→          setError('Connection issue. Restarting...')
   377→   377→          // Auto-recover after 3 seconds
   378→   378→          setTimeout(() => {
   379→   379→            resetState()
   380→   380→          }, 3000)
   381→   381→        }
   382→   382→      } else {
   383→   383→        loadingStartTimeRef.current = null
   384→   384→      }
   385→   385→    }
   386→   386→
   387→   387→    const interval = setInterval(checkStuckState, CHECK_INTERVAL_MS)
   388→   388→    return () => clearInterval(interval)
   389→   389→  }, [loading, loadingBoxes, resetState])
   390→   390→
   391→   391→  // Get available box sizes based on actual availability
   392→   392→  const BOX_SIZES = ALL_BOX_SIZES.filter(box => {

   395→   395→  }).map(box => {
   396→   396→    const availability = boxAvailability.find(a => a.code === box.code)
   397→   397→    return {
   398→   398→      ...box,
   399→   399→      available: availability?.available || 0,
   400→   400→    }
   401→   401→  })
   402→   402→

   404→   404→  useEffect(() => {
   405→   405→    const fetchBoxAvailability = async () => {
   406→   406→      try {
   407→   407→        const res = await kioskFetch('/api/boxes/availability', undefined, 15000)
   408→   408→        const data = await res.json()
   409→   409→        if (data.success) {
   410→   410→          setBoxAvailability(data.sizes)
   411→   411→        }
   412→   412→      } catch (err) {
   413→   413→        console.error('Failed to fetch box availability:', err)
   414→   414→        if (err instanceof Error && err.message.includes('timed out')) {
   415→   415→          console.error('[Kiosk] Box availability request timed out — server may be slow')
   416→   416→        }
   417→   417→      } finally {
   418→   418→        setLoadingBoxes(false)
   419→   419→      }
   420→   420→    }
   421→   421→    fetchBoxAvailability()
   422→   422→
   423→   423→    // Refresh box availability every 5 minutes to keep data fresh
   424→   424→    // and serve as a liveness check (if this stops running, the self-heal will detect it)
   425→   425→    const refreshInterval = setInterval(fetchBoxAvailability, 5 * 60 * 1000)
   426→   426→    return () => clearInterval(refreshInterval)
   427→   427→  }, [])
   428→   428→
   429→   429→  // Update last activity time
   430→   430→  const updateActivity = useCallback(() => {
   431→   431→    setLastActivity(Date.now())
   432→   432→    setShowTimeoutWarning(false)
   433→   433→    setSecondsRemaining(10)
   434→   434→  }, [])
   435→   435→
   436→   436→  // Auto-timeout effect (only active when not on home screen)
   437→   437→  useEffect(() => {
   438→   438→    // Skip timeout logic entirely on home screen
   439→   439→    if (view === "home") {
   440→   440→      setShowTimeoutWarning(false)
   441→   441→      return
   442→   442→    }
   443→   443→
   444→   444→    const checkTimeout = () => {
   445→   445→      const elapsed = Date.now() - lastActivity
   446→   446→      const remaining = AUTO_TIMEOUT_MS - elapsed
   447→   447→      
   448→   448→      if (remaining <= 10000 && remaining > 0 && !showTimeoutWarning) {
   449→   449→        setShowTimeoutWarning(true)
   450→   450→        setSecondsRemaining(Math.ceil(remaining / 1000))
   451→   451→      }
   452→   452→      
   453→   453→      if (elapsed >= AUTO_TIMEOUT_MS) {
   454→   454→        resetState()
   455→   455→      }
   456→   456→    }
   457→   457→
   458→   458→    const interval = setInterval(checkTimeout, 1000)
   459→   459→    return () => clearInterval(interval)
   460→   460→  }, [lastActivity, view, showTimeoutWarning, resetState])
   461→   461→
   462→   462→  // Countdown effect when warning is shown
   463→   463→  useEffect(() => {
   464→   464→    if (!showTimeoutWarning) return
   465→   465→    
   466→   466→    const countdownInterval = setInterval(() => {
   467→   467→      setSecondsRemaining(prev => {
   468→   468→        if (prev <= 1) {
   469→   469→          return 0
   470→   470→        }
   471→   471→        return prev - 1
   472→   472→      })
   473→   473→    }, 1000)
   474→   474→    
   475→   475→    return () => clearInterval(countdownInterval)
   476→   476→  }, [showTimeoutWarning])
   477→   477→
   478→   478→  // Handle screen touch/click to reset timeout
   479→   479→  useEffect(() => {
   480→   480→    const handleActivity = () => updateActivity()
   481→   481→    window.addEventListener('touchstart', handleActivity)
   482→   482→    window.addEventListener('click', handleActivity)
   483→   483→    return () => {
   484→   484→      window.removeEventListener('touchstart', handleActivity)
   485→   485→      window.removeEventListener('click', handleActivity)
   486→   486→    }
   487→   487→  }, [updateActivity])

   627→   627→  // Poll payment status with smart backoff
   628→   628→  const pollPaymentStatus = async (paymentId: string) => {
   629→   629→    const maxAttempts = 45 // ~6 minutes with backoff
   630→   630→    let attempts = 0
   631→   631→
   632→   632→    const poll = async () => {
   633→   633→      // Stop if component unmounted
   634→   634→      if (!isMountedRef.current) return
   635→   635→
   636→   636→      if (attempts >= maxAttempts) {
   637→   637→        setError("Payment timeout. Please try again.")
   638→   638→        return
   639→   639→      }
   640→   640→
   641→   641→      // Stop polling if tab is hidden
   642→   642→      if (document.visibilityState === 'hidden') {
   643→   643→        // Resume when visible again - track listener for cleanup
   644→   644→        const handleVisible = () => {
   645→   645→          if (document.visibilityState === 'visible') {
   646→   646→            document.removeEventListener('visibilitychange', handleVisible)
   647→   647→            if (visibilityHandlerRef.current === handleVisible) {
   648→   648→              visibilityHandlerRef.current = null
   649→   649→            }
   650→   650→            poll()
   651→   651→          }
   652→   652→        }
   653→   653→        visibilityHandlerRef.current = handleVisible
   654→   654→        document.addEventListener('visibilitychange', handleVisible)
   655→   655→        return
   656→   656→      }
   657→   657→
   658→   658→      attempts++
   659→   659→
   660→   660→      try {
   661→   661→        const res = await kioskFetch(`/api/kiosk/payment?paymentId=${paymentId}`, undefined, 10000)
   662→   662→        const data = await res.json()
   663→   663→
   664→   664→        if (!isMountedRef.current) return
   665→   665→
   666→   666→        if (data.status === "completed") {
   667→   667→          setCode(data.saveCode || "")
   668→   668→          // Go directly to dropoff-code view which shows the code
   669→   669→          setView("dropoff-code")
   670→   670→          setPaymentResult(null) // Clear payment result
   671→   671→          return
   672→   672→        }
   673→   673→
   674→   674→        if (data.status === "failed") {
   675→   675→          setError("Payment failed. Please try again.")
   676→   676→          return
   677→   677→        }
   678→   678→
   679→   679→        // Exponential backoff: 5s for first 10 attempts, then 8s, then 10s
   680→   680→        const delay = attempts < 10 ? 5000 : attempts < 25 ? 8000 : 10000
   681→   681→        pollingTimerRef.current = setTimeout(poll, delay)
   682→   682→      } catch {
   683→   683→        // Continue polling on error with longer delay
   684→   684→        pollingTimerRef.current = setTimeout(poll, 8000)
   685→   685→      }
   686→   686→    }
   687→   687→
   688→   688→    poll()
   689→   689→  }
