  1007→// POST handler - only for step=1 redirects
  1008→export async function POST(request: NextRequest) {
  1009→  const formData = await request.formData();
  1010→  const flow = formData.get('flow') as string;
  1011→  const step = formData.get('step') as string;
  1012→
  1013→  if (step === '1') {
  1014→    if (flow === 'dropoff') {
  1015→      return NextResponse.redirect(new URL('/kiosk-lite?action=dropoff', request.url), 303);
  1016→    }
  1017→    if (flow === 'pickup') {
  1018→      return NextResponse.redirect(new URL('/kiosk-lite?action=pickup', request.url), 303);
  1019→    }
  1020→  }
  1021→
  1022→  return NextResponse.redirect(new URL('/kiosk-lite', request.url), 303);
  1023→}
