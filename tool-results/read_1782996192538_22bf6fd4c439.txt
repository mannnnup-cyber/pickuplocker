    94→async function kioskFetch(url: string, options?: RequestInit, timeoutMs: number = FETCH_TIMEOUT_MS): Promise<Response> {
    95→  const controller = new AbortController()
    96→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
    97→  try {
    98→    const res = await fetch(url, { ...options, signal: controller.signal })
    99→    if (!res.ok) {
   100→      throw new Error(`HTTP ${res.status} from ${url}`)
   101→    }
   102→    return res
   103→  } catch (error) {
   104→    if (error instanceof DOMException && error.name === 'AbortError') {
   105→      throw new Error(`Request to ${url} timed out after ${timeoutMs}ms`)
   106→    }
   107→    throw error
   108→  } finally {
   109→    clearTimeout(timeoutId)
   110→  }
   111→}

   531→  // Poll payment status with smart backoff
   532→  const pollPaymentStatus = async (paymentId: string) => {
   533→    const maxAttempts = 45 // ~6 minutes with backoff
   534→    let attempts = 0
   535→
   536→    const poll = async () => {
   537→      // Stop if component unmounted
   538→      if (!isMountedRef.current) return
   539→
   540→      if (attempts >= maxAttempts) {
   541→        setError("Payment timeout. Please try again.")
   542→        return
   543→      }
   544→
   545→      // Stop polling if tab is hidden
   546→      if (document.visibilityState === 'hidden') {
   547→        // Resume when visible again - track listener for cleanup
   548→        const handleVisible = () => {
   549→          if (document.visibilityState === 'visible') {
   550→            document.removeEventListener('visibilitychange', handleVisible)
   551→            if (visibilityHandlerRef.current === handleVisible) {
   552→              visibilityHandlerRef.current = null
   553→            }
   554→            poll()
   555→          }
   556→        }
   557→        visibilityHandlerRef.current = handleVisible
   558→        document.addEventListener('visibilitychange', handleVisible)
   559→        return
   560→      }
   561→
   562→      attempts++
   563→
   564→      try {
   565→        const res = await kioskFetch(`/api/kiosk/payment?paymentId=${paymentId}`, undefined, 10000)
   566→        const data = await res.json()
   567→
   568→        if (!isMountedRef.current) return
   569→
   570→        if (data.status === "completed") {
   571→          setCode(data.saveCode || "")
   572→          // Go directly to dropoff-code view which shows the code
   573→          setView("dropoff-code")
   574→          setPaymentResult(null) // Clear payment result
   575→          return
   576→        }
   577→
   578→        if (data.status === "failed") {
   579→          setError("Payment failed. Please try again.")
   580→          return
   581→        }
   582→
   583→        // Exponential backoff: 5s for first 10 attempts, then 8s, then 10s
   584→        const delay = attempts < 10 ? 5000 : attempts < 25 ? 8000 : 10000
   585→        pollingTimerRef.current = setTimeout(poll, delay)
   586→      } catch {
   587→        // Continue polling on error with longer delay
   588→        pollingTimerRef.current = setTimeout(poll, 8000)
   589→      }
   590→    }
   591→
   592→    poll()
   593→  }

   768→              const delay = attempts < 10 ? 5000 : attempts < 25 ? 8000 : 10000
