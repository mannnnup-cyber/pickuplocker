   220→            case "portrait":
   221→                setRequestedOrientation(android.content.pm.ActivityInfo.SCREEN_ORIENTATION_SENSOR_PORTRAIT);

   225→                break;
   226→            case "auto":
   227→                setRequestedOrientation(android.content.pm.ActivityInfo.SCREEN_ORIENTATION_FULL_SENSOR);
   228→                break;
   229→            default:
   230→                setRequestedOrientation(android.content.pm.ActivityInfo.SCREEN_ORIENTATION_SENSOR_PORTRAIT);
   231→                break;
   232→        }
   233→    }

   348→    private void configureWebView() {
   349→        if (webView == null) return;
   350→        try {
   351→            WebSettings settings = webView.getSettings();
   352→            settings.setJavaScriptEnabled(true);
   353→            settings.setDomStorageEnabled(true);
   354→            settings.setAllowFileAccess(true);
   355→            settings.setAllowContentAccess(true);
   356→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
   357→                settings.setMixedContentMode(WebSettings.MIXED_CONTENT_ALWAYS_ALLOW);
   358→            }
   359→            settings.setCacheMode(WebSettings.LOAD_DEFAULT);
   360→            settings.setBuiltInZoomControls(false);
   361→            settings.setDisplayZoomControls(false);
   362→
   363→            String userAgent = settings.getUserAgentString();
   364→            settings.setUserAgentString(userAgent + " PickupKiosk/" + APP_VERSION);
   365→
   366→            try {
   367→                boolean isDebug = false;
   368→                try { isDebug = BuildConfig.DEBUG; } catch (Throwable ignored) {}
   369→                WebView.setWebContentsDebuggingEnabled(isDebug);
   370→            } catch (Throwable t) { Log.w(TAG, "WebView debugging: " + t.getMessage()); }
   371→
   372→            webView.setWebViewClient(new KioskWebViewClient());
   373→            webView.setWebChromeClient(new KioskWebChromeClient());
   374→            webView.loadUrl(getKioskUrl());
   375→        } catch (Throwable t) {
   376→            Log.e(TAG, "WebView config failed: " + t.getMessage(), t);
   377→            showErrorScreen("WebView configuration failed: " + t.getMessage());
   378→        }
   379→    }

   385→    private class KioskWebViewClient extends WebViewClient {
   386→        @Override
   387→        public void onPageFinished(WebView view, String url) {
   388→            super.onPageFinished(view, url);
   389→            if (url == null) return;
   390→            if (OFFLINE_URL.equals(url)) {
   391→                isShowingOfflinePage = true;
   392→                Log.w(TAG, "Showing offline page");
   393→            } else {
   394→                isShowingOfflinePage = false;
   395→                reconnectAttempts = 0;
   396→                Log.i(TAG, "Page loaded: " + url);
   397→            }
   398→        }
   399→
   400→        @Override
   401→        public void onReceivedError(WebView view, WebResourceRequest request, WebResourceError error) {
   402→            if (request != null && request.isForMainFrame()) {
   403→                if (!isShowingOfflinePage) showOfflinePage();
   404→            }
   405→        }
   406→
   407→        @Override
   408→        public void onReceivedError(WebView view, int errorCode, String description, String failingUrl) {
   409→            if (!isShowingOfflinePage) showOfflinePage();
   410→        }
   411→
   412→        @Override
   413→        public boolean shouldOverrideUrlLoading(WebView view, WebResourceRequest request) {
   414→            if (request == null) return false;
   415→            String url = request.getUrl().toString();
   416→            if (url == null) return true;
   417→            // Allow kiosk domain, backend, assets, and payment API
   418→            if (url.startsWith("https://pickuplocker.vercel.app") ||
   419→                url.startsWith("file:///android_asset/") ||
   420→                url.startsWith("https://api.dimepay.app")) {
   421→                return false;
   422→            }
   423→            Log.w(TAG, "Blocked navigation to: " + url);
   424→            return true;
   425→        }
   426→    }

   448→
   449→    private void showOfflinePage() {
   450→        isShowingOfflinePage = true;
   451→        if (webView != null && mainHandler != null) {
   452→            mainHandler.post(() -> {
   453→                if (webView != null && !isDestroyed) {
   454→                    try { webView.loadUrl(OFFLINE_URL); } catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   455→                }
   456→            });
   457→        }
   458→        scheduleReconnect();
   459→    }
   460→
   461→    private void scheduleReconnect() {
   462→        if (reconnectAttempts >= MAX_RECONNECT_ATTEMPTS) return;
   463→        reconnectAttempts++;
   464→        if (mainHandler != null) {
   465→            mainHandler.postDelayed(() -> {
   466→                if (isShowingOfflinePage && !isDestroyed) checkAndReload();
   467→            }, RECONNECT_DELAY_MS);
   468→        }
   469→    }
   470→
   471→    private void checkAndReload() {
   472→        if (!isNetworkAvailable()) { scheduleReconnect(); return; }
   473→        if (webView != null && mainHandler != null) {
   474→            mainHandler.post(() -> {
   475→                if (webView != null && !isDestroyed) {
   476→                    try { webView.loadUrl(isViewingBackend ? getBackendUrl() : getKioskUrl()); }
   477→                    catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   478→                }
   479→            });
   480→        }
   481→    }

   484→    // CONNECTIVITY MONITORING
   485→    // ============================================
   486→
   487→    private void registerConnectivityMonitoring() {
   488→        try {
   489→            ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   490→            if (cm == null) return;
   491→            NetworkRequest request = new NetworkRequest.Builder()
   492→                .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   493→                .build();
   494→            networkCallback = new ConnectivityManager.NetworkCallback() {
   495→                @Override
   496→                public void onAvailable(Network network) {
   497→                    if (isShowingOfflinePage && mainHandler != null) {
   498→                        mainHandler.postDelayed(() -> {
   499→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   500→                                reconnectAttempts = 0;
   501→                                mainHandler.post(() -> {
   502→                                    if (webView != null && !isDestroyed)
   503→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   504→                                });
   505→                            }
   506→                        }, CONNECTIVITY_CHECK_MS);
   507→                    }
   508→                }
   509→                @Override
   510→                public void onLost(Network network) { Log.w(TAG, "Network lost"); }
   511→                @Override
   512→                public void onCapabilitiesChanged(Network network, NetworkCapabilities caps) {
   513→                    if (caps == null) return;
   514→                    boolean ok = caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   515→                            && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED);
   516→                    if (ok && isShowingOfflinePage && mainHandler != null) {
   517→                        mainHandler.postDelayed(() -> {
   518→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   519→                                reconnectAttempts = 0;
   520→                                mainHandler.post(() -> {
   521→                                    if (webView != null && !isDestroyed)
   522→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   523→                                });
   524→                            }
   525→                        }, CONNECTIVITY_CHECK_MS);
   526→                    }
   527→                }
   528→            };
   529→            cm.registerNetworkCallback(request, networkCallback);
   530→        } catch (Throwable t) {
   531→            Log.e(TAG, "Network monitoring failed: " + t.getMessage());
   532→        }
   533→    }
   534→
   535→    private void unregisterConnectivityMonitoring() {
   536→        if (networkCallback != null) {
   537→            try {
   538→                ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   539→                if (cm != null) cm.unregisterNetworkCallback(networkCallback);
   540→            } catch (Throwable t) { Log.w(TAG, t.getMessage()); }
   541→        }
   542→    }
   543→
   544→    private boolean isNetworkAvailable() {
   545→        try {
   546→            ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   547→            if (cm == null) return false;
   548→            Network activeNetwork = cm.getActiveNetwork();
   549→            if (activeNetwork == null) return false;
   550→            NetworkCapabilities caps = cm.getNetworkCapabilities(activeNetwork);
   551→            return caps != null && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET);
   552→        } catch (Throwable t) { return true; }
   553→    }

   566→    // ============================================
   567→    // SECRET TRIGGER — 5-tap top-left corner
   568→    // ============================================
   569→
   570→    @Override
   571→    public boolean dispatchTouchEvent(MotionEvent ev) {
   572→        if (ev.getAction() == MotionEvent.ACTION_DOWN) {
   573→            try {
   574→                float density = getResources().getDisplayMetrics().density;
   575→                float hitAreaPx = SECRET_TAP_AREA_DP * density;
   576→                float x = ev.getRawX();
   577→                float y = ev.getRawY();
   578→                if (x < hitAreaPx && y < hitAreaPx) {
   579→                    long now = System.currentTimeMillis();
   580→                    if (now - lastSecretTapTime > SECRET_TAP_TIMEOUT_MS) {
   581→                        secretTapCount = 1;
   582→                    } else {
   583→                        secretTapCount++;
   584→                    }
   585→                    lastSecretTapTime = now;
   586→                    if (secretTapCount >= SECRET_TAP_COUNT) {
   587→                        secretTapCount = 0;
   588→                        Log.i(TAG, "Secret trigger activated");
   589→                        showPinDialog();
   590→                    }
   591→                }
   592→            } catch (Throwable t) {
   593→                Log.w(TAG, "Touch handler error: " + t.getMessage());
   594→            }
   595→        }
   596→        return super.dispatchTouchEvent(ev);
   597→    }

   603→    /**
   604→     * PIN entry dialog — titled "Device Code" (no hint about admin).
   605→     * Staff PIN → staff menu. Admin PIN → admin menu.
   606→     * 3 wrong attempts → 5 minute lockout.
   607→     */
   608→    private void showPinDialog() {
   609→        // Check lockout first
   610→        long lockoutUntil = getPrefs().getLong(KEY_PIN_LOCKOUT_UNTIL, 0);
   611→        if (System.currentTimeMillis() < lockoutUntil) {
   612→            long remainingSec = (lockoutUntil - System.currentTimeMillis()) / 1000;
   613→            Toast.makeText(this, "Locked. Try again in " + remainingSec + "s", Toast.LENGTH_LONG).show();
   614→            Log.w(TAG, "PIN entry locked out for " + remainingSec + " more seconds");
   615→            return;
   616→        }
   617→
   618→        try {
   619→            EditText pinInput = new EditText(this);
   620→            pinInput.setInputType(InputType.TYPE_CLASS_NUMBER | InputType.TYPE_NUMBER_VARIATION_PASSWORD);
   621→            pinInput.setHint("Enter device code");
   622→            pinInput.setMaxLines(1);
   623→
   624→            LinearLayout.LayoutParams lp = new LinearLayout.LayoutParams(
   625→                LinearLayout.LayoutParams.MATCH_PARENT, LinearLayout.LayoutParams.WRAP_CONTENT);
   626→            lp.setMargins(50, 20, 50, 20);
   627→            pinInput.setLayoutParams(lp);
   628→
   629→            LinearLayout container = new LinearLayout(this);
   630→            container.setOrientation(LinearLayout.VERTICAL);
   631→            container.setPadding(40, 20, 40, 20);
   632→            container.addView(pinInput);
   633→
   634→            new AlertDialog.Builder(this)
   635→                .setTitle("Device Code")
   636→                .setMessage("Enter code to continue.")
   637→                .setView(container)
   638→                .setPositiveButton("Confirm", (dialog, which) -> {
   639→                    String entered = pinInput.getText().toString().trim();
   640→                    if (entered.equals(getAdminPin())) {
   641→                        // Admin access — reset failed attempts
   642→                        resetPinAttempts();
   643→                        Log.i(TAG, "Admin access granted");
   644→                        showAdminMenu();
   645→                    } else if (entered.equals(getStaffPin())) {
   646→                        // Staff access — reset failed attempts
   647→                        resetPinAttempts();
   648→                        Log.i(TAG, "Staff access granted");
   649→                        showStaffMenu();
   650→                    } else {
   651→                        // Wrong PIN
   652→                        handleWrongPin();
   653→                    }
   654→                })
   655→                .setNegativeButton("Cancel", null)
   656→                .setCancelable(true)
   657→                .show();
   658→        } catch (Throwable t) {
   659→            Log.e(TAG, "PIN dialog failed: " + t.getMessage());
   660→        }
   661→    }
   662→
   663→    private void handleWrongPin() {
   664→        int attempts = getPrefs().getInt(KEY_PIN_FAILED_ATTEMPTS, 0) + 1;
   665→        getPrefs().edit().putInt(KEY_PIN_FAILED_ATTEMPTS, attempts).apply();
   666→        Log.w(TAG, "Wrong PIN attempt " + attempts + "/" + MAX_PIN_ATTEMPTS);
   667→
   668→        if (attempts >= MAX_PIN_ATTEMPTS) {
   669→            // Lock out for 5 minutes
   670→            long lockoutUntil = System.currentTimeMillis() + PIN_LOCKOUT_DURATION_MS;
   671→            getPrefs().edit()
   672→                .putLong(KEY_PIN_LOCKOUT_UNTIL, lockoutUntil)
   673→                .putInt(KEY_PIN_FAILED_ATTEMPTS, 0)
   674→                .apply();
   675→            Toast.makeText(this, "Too many attempts. Locked for 5 minutes.", Toast.LENGTH_LONG).show();
   676→        } else {
   677→            int remaining = MAX_PIN_ATTEMPTS - attempts;
   678→            new AlertDialog.Builder(this)
   679→                .setTitle("Incorrect Code")
   680→                .setMessage("The code you entered is incorrect.\n" + remaining + " attempt(s) remaining.")
   681→                .setPositiveButton("OK", null)
   682→                .show();
   683→        }
   684→    }

   691→    }
