   655→    private void checkWebViewHealth() {
   656→        if (webView == null) {
   657→            healthCheckFailCount++;
   658→            Log.w(TAG, "Health check: WebView is null (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   659→            if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   660→                Log.w(TAG, "Health check: Too many failures — attempting full recovery");
   661→                healthCheckFailCount = 0;
   662→                reloadKiosk();
   663→            }
   664→            return;
   665→        }
   666→
   667→        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.KITKAT) {
   668→            webView.evaluateJavascript("(function(){ try { var c = document.querySelector('.container'); return c ? 'ok' : 'empty'; } catch(e) { return 'error'; } })()", new android.webkit.ValueCallback<String>() {
   669→                @Override
   670→                public void onReceiveValue(String value) {
   671→                    if (value != null && (value.contains("ok") || value.contains("empty"))) {
   672→                        if (value.contains("empty")) {
   673→                            healthCheckFailCount++;
   674→                            Log.w(TAG, "Health check: Page content empty (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   675→                        } else {
   676→                            if (healthCheckFailCount > 0) {
   677→                                Log.i(TAG, "Health check: Page recovered (was " + healthCheckFailCount + " fails)");
   678→                            }
   679→                            healthCheckFailCount = 0;
   680→                        }
   681→                    } else {
   682→                        healthCheckFailCount++;
   683→                        Log.w(TAG, "Health check: No response or unexpected value: " + value + " (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   684→                    }
   685→
   686→                    if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   687→                        Log.w(TAG, "Health check: Threshold reached — force reloading kiosk");
   688→                        healthCheckFailCount = 0;
   689→                        reloadKiosk();
   690→                    }
   691→                }
   692→            });
   693→        } else {
   694→            // Pre-KitKat: just check offline status
   695→            if (isShowingOfflinePage) {
   696→                healthCheckFailCount++;
   697→                if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   698→                    healthCheckFailCount = 0;
   699→                    reloadKiosk();
   700→                }
   701→            } else {
   702→                healthCheckFailCount = 0;
   703→            }
   704→        }
   705→    }

   722→    @Override
   723→    public boolean dispatchTouchEvent(MotionEvent ev) {
   724→        if (ev.getAction() == MotionEvent.ACTION_DOWN) {
   725→            try {
   726→                float density = getResources().getDisplayMetrics().density;
   727→                float hitAreaPx = SECRET_TAP_AREA_DP * density;
   728→                float x = ev.getRawX();
   729→                float y = ev.getRawY();
   730→                if (x < hitAreaPx && y < hitAreaPx) {
   731→                    long now = System.currentTimeMillis();
   732→                    if (now - lastSecretTapTime > SECRET_TAP_TIMEOUT_MS) {
   733→                        secretTapCount = 1;
   734→                    } else {
   735→                        secretTapCount++;
   736→                    }
   737→                    lastSecretTapTime = now;
   738→                    if (secretTapCount >= SECRET_TAP_COUNT) {
   739→                        secretTapCount = 0;
   740→                        Log.i(TAG, "Secret trigger activated");
   741→                        showPinDialog();
   742→                    }
   743→                }
   744→            } catch (Throwable t) {
   745→                Log.w(TAG, "Touch handler error: " + t.getMessage());
   746→            }
   747→        }
   748→        return super.dispatchTouchEvent(ev);
   749→    }

   815→    private void handleWrongPin() {
   816→        int attempts = getPrefs().getInt(KEY_PIN_FAILED_ATTEMPTS, 0) + 1;
   817→        getPrefs().edit().putInt(KEY_PIN_FAILED_ATTEMPTS, attempts).apply();
   818→        Log.w(TAG, "Wrong PIN attempt " + attempts + "/" + MAX_PIN_ATTEMPTS);
   819→
   820→        if (attempts >= MAX_PIN_ATTEMPTS) {
   821→            // Lock out for 5 minutes
   822→            long lockoutUntil = System.currentTimeMillis() + PIN_LOCKOUT_DURATION_MS;
   823→            getPrefs().edit()
   824→                .putLong(KEY_PIN_LOCKOUT_UNTIL, lockoutUntil)
   825→                .putInt(KEY_PIN_FAILED_ATTEMPTS, 0)
   826→                .apply();
   827→            Toast.makeText(this, "Too many attempts. Locked for 5 minutes.", Toast.LENGTH_LONG).show();
   828→        } else {
   829→            int remaining = MAX_PIN_ATTEMPTS - attempts;
   830→            new AlertDialog.Builder(this)
   831→                .setTitle("Incorrect Code")
   832→                .setMessage("The code you entered is incorrect.\n" + remaining + " attempt(s) remaining.")
   833→                .setPositiveButton("OK", null)
   834→                .show();
   835→        }
   836→    }
