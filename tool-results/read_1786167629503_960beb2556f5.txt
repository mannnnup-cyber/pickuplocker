   404→   404→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
   405→   405→                settings.setMixedContentMode(WebSettings.MIXED_CONTENT_ALWAYS_ALLOW);
   406→   406→            }
   407→   407→            settings.setCacheMode(WebSettings.LOAD_DEFAULT);
   408→   408→            settings.setBuiltInZoomControls(false);
   409→   409→            settings.setDisplayZoomControls(false);
   410→   410→
   411→   411→            String userAgent = settings.getUserAgentString();
   412→   412→            settings.setUserAgentString(userAgent + " PickupKiosk/" + APP_VERSION);
   413→   413→
   414→   414→            try {
   415→   415→                boolean isDebug = false;
   416→   416→                try { isDebug = BuildConfig.DEBUG; } catch (Throwable ignored) {}
   417→   417→                WebView.setWebContentsDebuggingEnabled(isDebug);
   418→   418→            } catch (Throwable t) { Log.w(TAG, "WebView debugging: " + t.getMessage()); }
   419→   419→
   420→   420→            webView.setWebViewClient(new KioskWebViewClient());
   421→   421→            webView.setWebChromeClient(new KioskWebChromeClient());
   422→   422→            webView.loadUrl(getKioskUrl());

   424→   424→            Log.e(TAG, "WebView config failed: " + t.getMessage(), t);
   425→   425→            showErrorScreen("WebView configuration failed: " + t.getMessage());
   426→   426→        }

   433→   433→    private class KioskWebViewClient extends WebViewClient {
   434→   434→        @Override
   435→   435→        public void onPageFinished(WebView view, String url) {
   436→   436→            super.onPageFinished(view, url);
   437→   437→            if (url == null) return;
   438→   438→            if (OFFLINE_URL.equals(url)) {
   439→   439→                isShowingOfflinePage = true;
   440→   440→                Log.w(TAG, "Showing offline page");
   441→   441→            } else {
   442→   442→                isShowingOfflinePage = false;
   443→   443→                reconnectAttempts = 0;
   444→   444→                healthCheckFailCount = 0; // Reset health on successful page load
   445→   445→                Log.i(TAG, "Page loaded: " + url);
   446→   446→            }
   447→   447→        }
   448→   448→
   449→   449→        @Override
   450→   450→        public void onReceivedError(WebView view, WebResourceRequest request, WebResourceError error) {
   451→   451→            if (request != null && request.isForMainFrame()) {
   452→   452→                if (!isShowingOfflinePage) showOfflinePage();
   453→   453→            }
   454→   454→        }
   455→   455→
   456→   456→        @Override
   457→   457→        public void onReceivedError(WebView view, int errorCode, String description, String failingUrl) {
   458→   458→            if (!isShowingOfflinePage) showOfflinePage();
   459→   459→        }
   460→   460→
   461→   461→        @Override
   462→   462→        public boolean shouldOverrideUrlLoading(WebView view, WebResourceRequest request) {
   463→   463→            if (request == null) return false;
   464→   464→            String url = request.getUrl().toString();
   465→   465→            if (url == null) return true;
   466→   466→            // Allow kiosk domain, backend, assets, and payment API
   467→   467→            if (url.startsWith("https://pickuplocker.vercel.app") ||
   468→   468→                url.startsWith("file:///android_asset/") ||
   469→   469→                url.startsWith("https://api.dimepay.app")) {
   470→   470→                return false;
   471→   471→            }
   472→   472→            Log.w(TAG, "Blocked navigation to: " + url);
   473→   473→            return true;
   474→   474→        }
   475→   475→
   476→   476→        /**
   477→   477→         * CRITICAL: Called when the WebView renderer process crashes (OOM kill, etc.)
   478→   478→         * This is the #1 cause of white screens — Android kills the renderer but
   479→   479→         * the Activity is still alive. Without this handler, the WebView shows blank.
   480→   480→         */
   481→   481→        @Override
   482→   482→        public boolean onRenderProcessGone(WebView view, android.webkit.RenderProcessGoneDetail detail) {
   483→   483→            Log.e(TAG, "WebView renderer process GONE! Did crash: " + detail.didCrash()
   484→   484→                + ". Rebuilding WebView immediately.");
   485→   485→
   486→   486→            if (isDestroyed) return true;
   487→   487→
   488→   488→            // Destroy the old WebView completely
   489→   489→            if (webView != null) {
   490→   490→                try {
   491→   491→                    webView.stopLoading();
   492→   492→                    webView.loadUrl("about:blank");
   493→   493→                    webView.clearCache(true);
   494→   494→                    webView.clearHistory();
   495→   495→                    ViewGroup parent = (ViewGroup) webView.getParent();
   496→   496→                    if (parent != null) parent.removeView(webView);
   497→   497→                    webView.destroy();
   498→   498→                } catch (Throwable t) {
   499→   499→                    Log.e(TAG, "Error destroying dead WebView: " + t.getMessage());
   500→   500→                }
   501→   501→                webView = null;
   502→   502→            }
   503→   503→
   504→   504→            // Rebuild a fresh WebView
   505→   505→            mainHandler.post(() -> {
   506→   506→                if (isDestroyed) return;
   507→   507→                try {
   508→   508→                    webView = createWebViewSafely();
   509→   509→                    if (webView != null) {
   510→   510→                        setContentView(webView);
   511→   511→                        hideSystemUI();
   512→   512→                        configureWebView();
   513→   513→                        webView.loadUrl(getKioskUrl());
   514→   514→                        isShowingOfflinePage = false;
   515→   515→                        reconnectAttempts = 0;
   516→   516→                        healthCheckFailCount = 0;
   517→   517→                        Log.i(TAG, "WebView rebuilt after renderer crash — kiosk reloaded");
   518→   518→                    } else {
   519→   519→                        showErrorScreen("WebView failed to restart. Please reboot the device.");
   520→   520→                    }
   521→   521→                } catch (Throwable t) {
   522→   522→                    Log.e(TAG, "Failed to rebuild WebView: " + t.getMessage());
   523→   523→                    showErrorScreen("Critical error. Please reboot the device.");
   524→   524→                }
   525→   525→            });
   526→   526→
   527→   527→            return true; // We handled it — don't let WebView do its default (which does nothing)
   528→   528→        }
   529→   529→    }

   612→   612→                @Override

   618→   618→                            && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED);

   620→   620→                        mainHandler.postDelayed(() -> {
   621→   621→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   622→   622→                                reconnectAttempts = 0;
   623→   623→                                mainHandler.post(() -> {
   624→   624→                                    if (webView != null && !isDestroyed)
   625→   625→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   626→   626→                                });
   627→   627→                            }
   628→   628→                        }, CONNECTIVITY_CHECK_MS);
   629→   629→                    }
   630→   630→                }
   631→   631→            };
   632→   632→            cm.registerNetworkCallback(request, networkCallback);
   633→   633→        } catch (Throwable t) {
   634→   634→            Log.e(TAG, "Network monitoring failed: " + t.getMessage());
   635→   635→        }
   636→   636→    }

   659→   659→    // WEBVIEW HEALTH MONITORING — Detect frozen/white screens
   660→   660→    // ============================================
   661→   661→
   662→   662→    /**
   663→   663→     * Starts periodic health checks on the WebView.
   664→   664→     * If the WebView appears unresponsive or has blank content for
   665→   665→     * multiple consecutive checks, force a reload.
   666→   666→     */
   667→   667→    private void startHealthMonitoring() {
   668→   668→        healthCheckRunnable = new Runnable() {
   669→   669→            @Override
   670→   670→            public void run() {
   671→   671→                if (!isDestroyed) {
   672→   672→                    checkWebViewHealth();
   673→   673→                    mainHandler.postDelayed(this, HEALTH_CHECK_INTERVAL_MS);
   674→   674→                }
   675→   675→            }
   676→   676→        };
   677→   677→        mainHandler.postDelayed(healthCheckRunnable, HEALTH_CHECK_INTERVAL_MS);
   678→   678→        Log.i(TAG, "WebView health monitoring started (interval: " + HEALTH_CHECK_INTERVAL_MS + "ms)");
   679→   679→    }
   680→   680→
   681→   681→    private void stopHealthMonitoring() {
   682→   682→        if (healthCheckRunnable != null) {
   683→   683→            mainHandler.removeCallbacks(healthCheckRunnable);
   684→   684→            healthCheckRunnable = null;
   685→   685→            Log.i(TAG, "WebView health monitoring stopped");
   686→   686→        }
   687→   687→    }
   688→   688→
   689→   689→    /**
   690→   690→     * Check if the WebView is healthy by evaluating its content.
   691→   691→     * If the page appears blank or frozen for several consecutive checks,
   692→   692→     * force a reload to recover from white screen issues.
   693→   693→     */
   694→   694→    private void checkWebViewHealth() {
   695→   695→        if (webView == null) {
   696→   696→            healthCheckFailCount++;
   697→   697→            Log.w(TAG, "Health check: WebView is null (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   698→   698→            if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   699→   699→                Log.w(TAG, "Health check: Too many failures — attempting full recovery");
   700→   700→                healthCheckFailCount = 0;
   701→   701→                reloadKiosk();
   702→   702→            }
   703→   703→            return;
   704→   704→        }
   705→   705→
   706→   706→        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.KITKAT) {
   707→   707→            webView.evaluateJavascript("(function(){ try { var c = document.querySelector('.container'); return c ? 'ok' : 'empty'; } catch(e) { return 'error'; } })()", new android.webkit.ValueCallback<String>() {
   708→   708→                @Override
   709→   709→                public void onReceiveValue(String value) {
   710→   710→                    if (value != null && (value.contains("ok") || value.contains("empty"))) {
   711→   711→                        if (value.contains("empty")) {
   712→   712→                            healthCheckFailCount++;
   713→   713→                            Log.w(TAG, "Health check: Page content empty (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   714→   714→                        } else {
   715→   715→                            if (healthCheckFailCount > 0) {
   716→   716→                                Log.i(TAG, "Health check: Page recovered (was " + healthCheckFailCount + " fails)");
   717→   717→                            }
   718→   718→                            healthCheckFailCount = 0;
   719→   719→                        }
   720→   720→                    } else {
   721→   721→                        healthCheckFailCount++;
   722→   722→                        Log.w(TAG, "Health check: No response or unexpected value: " + value + " (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   723→   723→                    }
   724→   724→
   725→   725→                    if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   726→   726→                        Log.w(TAG, "Health check: Threshold reached — force reloading kiosk");
   727→   727→                        healthCheckFailCount = 0;
   728→   728→                        reloadKiosk();
   729→   729→                    }
   730→   730→                }
   731→   731→            });
   732→   732→        } else {
   733→   733→            // Pre-KitKat: just check offline status
   734→   734→            if (isShowingOfflinePage) {
   735→   735→                healthCheckFailCount++;
   736→   736→                if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   737→   737→                    healthCheckFailCount = 0;
   738→   738→                    reloadKiosk();
   739→   739→                }
   740→   740→            } else {
   741→   741→                healthCheckFailCount = 0;
   742→   742→            }
   743→   743→        }
   744→   744→    }

  1258→  1258→    /**
  1259→  1259→     * Schedules periodic update checks.
  1260→  1260→     * First check happens 30s after startup, then every 4 hours.
  1261→  1261→     */
  1262→  1262→    private void scheduleUpdateChecks() {
  1263→  1263→        updateCheckRunnable = new Runnable() {
  1264→  1264→            @Override
  1265→  1265→            public void run() {
  1266→  1266→                if (!isDestroyed) {
  1267→  1267→                    checkForUpdatesInBackground();
  1268→  1268→                    mainHandler.postDelayed(this, UPDATE_CHECK_INTERVAL_MS);
  1269→  1269→                }
  1270→  1270→            }
  1271→  1271→        };
  1272→  1272→        // Delay first check — don't slow down startup
  1273→  1273→        mainHandler.postDelayed(updateCheckRunnable, UPDATE_CHECK_DELAY_MS);
  1274→  1274→        Log.i(TAG, "Update checks scheduled (first in " + UPDATE_CHECK_DELAY_MS + "ms, then every " + UPDATE_CHECK_INTERVAL_MS + "ms)");
  1275→  1275→    }
  1276→  1276→
  1277→  1277→    private void stopUpdateChecks() {
  1278→  1278→        if (updateCheckRunnable != null) {
  1279→  1279→            mainHandler.removeCallbacks(updateCheckRunnable);
  1280→  1280→            updateCheckRunnable = null;
  1281→  1281→            Log.i(TAG, "Update checks stopped");
  1282→  1282→        }
  1283→  1283→    }
  1284→  1284→
  1285→  1285→    /**
  1286→  1286→     * Checks for updates in the background (no UI).
  1287→  1287→     * If an update is found and forceUpdate is true, shows a blocking dialog.
  1288→  1288→     * Otherwise, stores the info for the admin to install later.
  1289→  1289→     */
  1290→  1290→    private void checkForUpdatesInBackground() {
  1291→  1291→        new Thread(() -> {
  1292→  1292→            try {
  1293→  1293→                String baseUrl = getKioskUrl();
  1294→  1294→                // Extract origin from kiosk URL
  1295→  1295→                String origin = baseUrl;
  1296→  1296→                if (baseUrl.contains("//")) {
  1297→  1297→                    int pathStart = baseUrl.indexOf("//", baseUrl.indexOf("//") + 1);
  1298→  1298→                    if (pathStart > 0) {
  1299→  1299→                        // Find the first / after the host
  1300→  1300→                        int slashIndex = baseUrl.indexOf("/", baseUrl.indexOf("//") + 2);
  1301→  1301→                        if (slashIndex > 0) {
  1302→  1302→                            origin = baseUrl.substring(0, slashIndex);
  1303→  1303→                        } else {
  1304→  1304→                            origin = baseUrl;
  1305→  1305→                        }
  1306→  1306→                    }
  1307→  1307→                }
  1308→  1308→                String checkUrl = origin + UPDATE_CHECK_URL;
  1309→  1309→                Log.i(TAG, "Checking for updates: " + checkUrl);
  1310→  1310→
  1311→  1311→                URL url = new URL(checkUrl);
  1312→  1312→                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
  1313→  1313→                conn.setConnectTimeout(15000);
  1314→  1314→                conn.setReadTimeout(15000);
  1315→  1315→                conn.setRequestMethod("GET");
  1316→  1316→                conn.setRequestProperty("User-Agent", "PickupKiosk/" + APP_VERSION);
  1317→  1317→
  1318→  1318→                int responseCode = conn.getResponseCode();
  1319→  1319→                if (responseCode != 200) {
  1320→  1320→                    Log.w(TAG, "Update check failed: HTTP " + responseCode);
  1321→  1321→                    return;
  1322→  1322→                }
  1323→  1323→
  1324→  1324→                BufferedReader reader = new BufferedReader(new InputStreamReader(conn.getInputStream()));
  1325→  1325→                StringBuilder response = new StringBuilder();
  1326→  1326→                String line;
  1327→  1327→                while ((line = reader.readLine()) != null) {
  1328→  1328→                    response.append(line);
  1329→  1329→                }
  1330→  1330→                reader.close();
  1331→  1331→
  1332→  1332→                String jsonResponse = response.toString();
  1333→  1333→                Log.i(TAG, "Update check response: " + jsonResponse);
  1334→  1334→
  1335→  1335→                // Parse JSON manually (no Gson/Jackson in bare build)
  1336→  1336→                String serverVersion = extractJsonString(jsonResponse, "version");
  1337→  1337→                int serverVersionCode = extractJsonInt(jsonResponse, "versionCode");
  1338→  1338→                String apkUrl = extractJsonString(jsonResponse, "apkUrl");
  1339→  1339→                String checksum = extractJsonString(jsonResponse, "checksum");
  1340→  1340→                String changelog = extractJsonString(jsonResponse, "changelog");
  1341→  1341→                boolean forceUpdate = extractJsonBoolean(jsonResponse, "forceUpdate");
  1342→  1342→
  1343→  1343→                if (serverVersion == null || serverVersion.isEmpty()) {
  1344→  1344→                    Log.w(TAG, "Update check: no version info in response");
  1345→  1345→                    return;
  1346→  1346→                }
  1347→  1347→
  1348→  1348→                // Store update info
  1349→  1349→                getPrefs().edit()
  1350→  1350→                    .putString(KEY_AVAILABLE_VERSION, serverVersion)
  1351→  1351→                    .putInt(KEY_AVAILABLE_VERSION_CODE, serverVersionCode)

  1782→  1782→    private String computeSHA256(File file) throws Exception {
  1783→  1783→        MessageDigest digest = MessageDigest.getInstance("SHA-256");
  1784→  1784→        FileInputStream fis = new FileInputStream(file);
  1785→  1785→        byte[] buffer = new byte[8192];
  1786→  1786→        int bytesRead;
  1787→  1787→        while ((bytesRead = fis.read(buffer)) != -1) {
  1788→  1788→            digest.update(buffer, 0, bytesRead);
  1789→  1789→        }
  1790→  1790→        fis.close();
  1791→  1791→
  1792→  1792→        byte[] hash = digest.digest();
  1793→  1793→        StringBuilder hexString = new StringBuilder();
  1794→  1794→        for (byte b : hash) {
  1795→  1795→            String hex = Integer.toHexString(0xff & b);
  1796→  1796→            if (hex.length() == 1) hexString.append('0');
  1797→  1797→            hexString.append(hex);
  1798→  1798→        }
  1799→  1799→        return hexString.toString();
  1800→  1800→    }
