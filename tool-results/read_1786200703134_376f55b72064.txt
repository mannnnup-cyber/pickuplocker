   400→            settings.setJavaScriptEnabled(true);

   412→            settings.setUserAgentString(userAgent + " PickupKiosk/" + APP_VERSION);

   414→            try {
   415→                boolean isDebug = false;
   416→                try { isDebug = BuildConfig.DEBUG; } catch (Throwable ignored) {}
   417→                WebView.setWebContentsDebuggingEnabled(isDebug);
   418→            } catch (Throwable t) { Log.w(TAG, "WebView debugging: " + t.getMessage()); }

   420→            webView.setWebViewClient(new KioskWebViewClient());

   433→    private class KioskWebViewClient extends WebViewClient {
   434→        @Override
   435→        public void onPageFinished(WebView view, String url) {
   436→            super.onPageFinished(view, url);
   437→            if (url == null) return;
   438→            if (OFFLINE_URL.equals(url)) {
   439→                isShowingOfflinePage = true;
   440→                Log.w(TAG, "Showing offline page");
   441→            } else {
   442→                isShowingOfflinePage = false;
   443→                reconnectAttempts = 0;
   444→                healthCheckFailCount = 0; // Reset health on successful page load
   445→                Log.i(TAG, "Page loaded: " + url);
   446→            }
   447→        }
   448→
   449→        @Override
   450→        public void onReceivedError(WebView view, WebResourceRequest request, WebResourceError error) {
   451→            if (request != null && request.isForMainFrame()) {
   452→                if (!isShowingOfflinePage) showOfflinePage();
   453→            }
   454→        }
   455→
   456→        @Override
   457→        public void onReceivedError(WebView view, int errorCode, String description, String failingUrl) {
   458→            if (!isShowingOfflinePage) showOfflinePage();
   459→        }
   460→
   461→        @Override
   462→        public boolean shouldOverrideUrlLoading(WebView view, WebResourceRequest request) {
   463→            if (request == null) return false;
   464→            String url = request.getUrl().toString();
   465→            if (url == null) return true;
   466→            // Allow kiosk domain, backend, assets, and payment API
   467→            if (url.startsWith("https://pickuplocker.vercel.app") ||
   468→                url.startsWith("file:///android_asset/") ||
   469→                url.startsWith("https://api.dimepay.app")) {
   470→                return false;
   471→            }
   472→            Log.w(TAG, "Blocked navigation to: " + url);
   473→            return true;
   474→        }
   475→
   476→        /**
   477→         * CRITICAL: Called when the WebView renderer process crashes (OOM kill, etc.)
   478→         * This is the #1 cause of white screens — Android kills the renderer but
   479→         * the Activity is still alive. Without this handler, the WebView shows blank.
   480→         */
   481→        @Override
   482→        public boolean onRenderProcessGone(WebView view, android.webkit.RenderProcessGoneDetail detail) {
   483→            Log.e(TAG, "WebView renderer process GONE! Did crash: " + detail.didCrash()
   484→                + ". Rebuilding WebView immediately.");
   485→
   486→            if (isDestroyed) return true;
   487→
   488→            // Destroy the old WebView completely
   489→            if (webView != null) {
   490→                try {
   491→                    webView.stopLoading();
   492→                    webView.loadUrl("about:blank");
   493→                    webView.clearCache(true);
   494→                    webView.clearHistory();
   495→                    ViewGroup parent = (ViewGroup) webView.getParent();
   496→                    if (parent != null) parent.removeView(webView);
   497→                    webView.destroy();
   498→                } catch (Throwable t) {
   499→                    Log.e(TAG, "Error destroying dead WebView: " + t.getMessage());
   500→                }
   501→                webView = null;
   502→            }
   503→
   504→            // Rebuild a fresh WebView
   505→            mainHandler.post(() -> {
   506→                if (isDestroyed) return;
   507→                try {
   508→                    webView = createWebViewSafely();
   509→                    if (webView != null) {
   510→                        setContentView(webView);
   511→                        hideSystemUI();
   512→                        configureWebView();
   513→                        webView.loadUrl(getKioskUrl());
   514→                        isShowingOfflinePage = false;
   515→                        reconnectAttempts = 0;
   516→                        healthCheckFailCount = 0;
   517→                        Log.i(TAG, "WebView rebuilt after renderer crash — kiosk reloaded");
   518→                    } else {
   519→                        showErrorScreen("WebView failed to restart. Please reboot the device.");
   520→                    }
   521→                } catch (Throwable t) {
   522→                    Log.e(TAG, "Failed to rebuild WebView: " + t.getMessage());
   523→                    showErrorScreen("Critical error. Please reboot the device.");
   524→                }
   525→            });
   526→
   527→            return true; // We handled it — don't let WebView do its default (which does nothing)
   528→        }
   529→    }

   548→    // ============================================
   549→    // OFFLINE & RECONNECT
   550→    // ============================================
   551→
   552→    private void showOfflinePage() {
   553→        isShowingOfflinePage = true;
   554→        if (webView != null && mainHandler != null) {
   555→            mainHandler.post(() -> {
   556→                if (webView != null && !isDestroyed) {
   557→                    try { webView.loadUrl(OFFLINE_URL); } catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   558→                }
   559→            });
   560→        }
   561→        scheduleReconnect();
   562→    }
   563→
   564→    private void scheduleReconnect() {
   565→        if (reconnectAttempts >= MAX_RECONNECT_ATTEMPTS) return;
   566→        reconnectAttempts++;
   567→        if (mainHandler != null) {
   568→            mainHandler.postDelayed(() -> {
   569→                if (isShowingOfflinePage && !isDestroyed) checkAndReload();
   570→            }, RECONNECT_DELAY_MS);
   571→        }
   572→    }
   573→
   574→    private void checkAndReload() {
   575→        if (!isNetworkAvailable()) { scheduleReconnect(); return; }
   576→        if (webView != null && mainHandler != null) {
   577→            mainHandler.post(() -> {
   578→                if (webView != null && !isDestroyed) {
   579→                    try { webView.loadUrl(isViewingBackend ? getBackendUrl() : getKioskUrl()); }
   580→                    catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   581→                }
   582→            });
   583→        }
   584→    }

   587→    // CONNECTIVITY MONITORING
   588→    // ============================================

   590→    private void registerConnectivityMonitoring() {
   591→        try {
   592→            ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   593→            if (cm == null) return;
   594→            NetworkRequest request = new NetworkRequest.Builder()
   595→                .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   596→                .build();
   597→            networkCallback = new ConnectivityManager.NetworkCallback() {
   598→                @Override
   599→                public void onAvailable(Network network) {
   600→                    if (isShowingOfflinePage && mainHandler != null) {
   601→                        mainHandler.postDelayed(() -> {
   602→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   603→                                reconnectAttempts = 0;
   604→                                mainHandler.post(() -> {
   605→                                    if (webView != null && !isDestroyed)
   606→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   607→                                });
   608→                            }
   609→                        }, CONNECTIVITY_CHECK_MS);
   610→                    }
   611→                }
   612→                @Override
   613→                public void onLost(Network network) { Log.w(TAG, "Network lost"); }
   614→                @Override
   615→                public void onCapabilitiesChanged(Network network, NetworkCapabilities caps) {
   616→                    if (caps == null) return;
   617→                    boolean ok = caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   618→                            && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED);
   619→                    if (ok && isShowingOfflinePage && mainHandler != null) {
   620→                        mainHandler.postDelayed(() -> {
   621→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   622→                                reconnectAttempts = 0;
   623→                                mainHandler.post(() -> {
   624→                                    if (webView != null && !isDestroyed)
   625→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   626→                                });
   627→                            }
   628→                        }, CONNECTIVITY_CHECK_MS);
   629→                    }
   630→                }
   631→            };
   632→            cm.registerNetworkCallback(request, networkCallback);
   633→        } catch (Throwable t) {
   634→            Log.e(TAG, "Network monitoring failed: " + t.getMessage());
   635→        }
   636→    }
   637→
   638→    private void unregisterConnectivityMonitoring() {
   639→        if (networkCallback != null) {
   640→            try {
   641→                ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   642→                if (cm != null) cm.unregisterNetworkCallback(networkCallback);
   643→            } catch (Throwable t) { Log.w(TAG, t.getMessage()); }
   644→        }
   645→    }

   647→    private boolean isNetworkAvailable() {
   648→        try {
   649→            ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   650→            if (cm == null) return false;
   651→            Network activeNetwork = cm.getActiveNetwork();
   652→            if (activeNetwork == null) return false;
   653→            NetworkCapabilities caps = cm.getNetworkCapabilities(activeNetwork);
   654→            return caps != null && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET);
   655→        } catch (Throwable t) { return true; }
   656→    }

   658→    // ============================================
   659→    // WEBVIEW HEALTH MONITORING — Detect frozen/white screens
   660→    // ============================================
   661→
   662→    /**
   663→     * Starts periodic health checks on the WebView.
   664→     * If the WebView appears unresponsive or has blank content for
   665→     * multiple consecutive checks, force a reload.
   666→     */
   667→    private void startHealthMonitoring() {
   668→        healthCheckRunnable = new Runnable() {
   669→            @Override
   670→            public void run() {
   671→                if (!isDestroyed) {
   672→                    checkWebViewHealth();
   673→                    mainHandler.postDelayed(this, HEALTH_CHECK_INTERVAL_MS);
   674→                }
   675→            }
   676→        };
   677→        mainHandler.postDelayed(healthCheckRunnable, HEALTH_CHECK_INTERVAL_MS);
   678→        Log.i(TAG, "WebView health monitoring started (interval: " + HEALTH_CHECK_INTERVAL_MS + "ms)");
   679→    }
   680→
   681→    private void stopHealthMonitoring() {
   682→        if (healthCheckRunnable != null) {
   683→            mainHandler.removeCallbacks(healthCheckRunnable);
   684→            healthCheckRunnable = null;
   685→            Log.i(TAG, "WebView health monitoring stopped");
   686→        }
   687→    }
   688→
   689→    /**
   690→     * Check if the WebView is healthy by evaluating its content.
   691→     * If the page appears blank or frozen for several consecutive checks,
   692→     * force a reload to recover from white screen issues.
   693→     */
   694→    private void checkWebViewHealth() {
   695→        if (webView == null) {
   696→            healthCheckFailCount++;
   697→            Log.w(TAG, "Health check: WebView is null (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   698→            if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   699→                Log.w(TAG, "Health check: Too many failures — attempting full recovery");
   700→                healthCheckFailCount = 0;
   701→                reloadKiosk();
   702→            }
   703→            return;
   704→        }
   705→
   706→        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.KITKAT) {
   707→            webView.evaluateJavascript("(function(){ try { var c = document.querySelector('.container'); return c ? 'ok' : 'empty'; } catch(e) { return 'error'; } })()", new android.webkit.ValueCallback<String>() {
   708→                @Override
   709→                public void onReceiveValue(String value) {
   710→                    if (value != null && (value.contains("ok") || value.contains("empty"))) {
   711→                        if (value.contains("empty")) {
   712→                            healthCheckFailCount++;
   713→                            Log.w(TAG, "Health check: Page content empty (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   714→                        } else {
   715→                            if (healthCheckFailCount > 0) {
   716→                                Log.i(TAG, "Health check: Page recovered (was " + healthCheckFailCount + " fails)");
   717→                            }
   718→                            healthCheckFailCount = 0;
   719→                        }
   720→                    } else {
   721→                        healthCheckFailCount++;
   722→                        Log.w(TAG, "Health check: No response or unexpected value: " + value + " (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   723→                    }
   724→
   725→                    if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   726→                        Log.w(TAG, "Health check: Threshold reached — force reloading kiosk");
   727→                        healthCheckFailCount = 0;
   728→                        reloadKiosk();
   729→                    }
   730→                }
   731→            });
   732→        } else {
   733→            // Pre-KitKat: just check offline status
   734→            if (isShowingOfflinePage) {
   735→                healthCheckFailCount++;
   736→                if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   737→                    healthCheckFailCount = 0;
   738→                    reloadKiosk();
   739→                }
   740→            } else {
   741→                healthCheckFailCount = 0;
   742→            }
   743→        }
   744→    }

   758→    // SECRET TRIGGER — 5-tap top-left corner
   759→    // ============================================
   760→
   761→    @Override
   762→    public boolean dispatchTouchEvent(MotionEvent ev) {
   763→        if (ev.getAction() == MotionEvent.ACTION_DOWN) {
   764→            try {
   765→                float density = getResources().getDisplayMetrics().density;
   766→                float hitAreaPx = SECRET_TAP_AREA_DP * density;
   767→                float x = ev.getRawX();
   768→                float y = ev.getRawY();
   769→                if (x < hitAreaPx && y < hitAreaPx) {
   770→                    long now = System.currentTimeMillis();
   771→                    if (now - lastSecretTapTime > SECRET_TAP_TIMEOUT_MS) {
   772→                        secretTapCount = 1;
   773→                    } else {
   774→                        secretTapCount++;
   775→                    }
   776→                    lastSecretTapTime = now;
   777→                    if (secretTapCount >= SECRET_TAP_COUNT) {
   778→                        secretTapCount = 0;
   779→                        Log.i(TAG, "Secret trigger activated");
   780→                        showPinDialog();
   781→                    }
   782→                }
   783→            } catch (Throwable t) {
   784→                Log.w(TAG, "Touch handler error: " + t.getMessage());
   785→            }
   786→        }
   787→        return super.dispatchTouchEvent(ev);
   788→    }

   808→
   809→        try {
   810→            EditText pinInput = new EditText(this);
   811→            pinInput.setInputType(InputType.TYPE_CLASS_NUMBER | InputType.TYPE_NUMBER_VARIATION_PASSWORD);

   820→            LinearLayout container = new LinearLayout(this);
   821→            container.setOrientation(LinearLayout.VERTICAL);
   822→            container.setPadding(40, 20, 40, 20);

   824→
   825→            new AlertDialog.Builder(this)

   837→                        // Staff access — reset failed attempts
   838→                        resetPinAttempts();
   839→                        Log.i(TAG, "Staff access granted");
   840→                        showStaffMenu();
   841→                    } else {
   842→                        // Wrong PIN
   843→                        handleWrongPin();
   844→                    }
   845→                })
   846→                .setNegativeButton("Cancel", null)
   847→                .setCancelable(true)
   848→                .show();

   854→    private void handleWrongPin() {
   855→        int attempts = getPrefs().getInt(KEY_PIN_FAILED_ATTEMPTS, 0) + 1;
   856→        getPrefs().edit().putInt(KEY_PIN_FAILED_ATTEMPTS, attempts).apply();
   857→        Log.w(TAG, "Wrong PIN attempt " + attempts + "/" + MAX_PIN_ATTEMPTS);
   858→
   859→        if (attempts >= MAX_PIN_ATTEMPTS) {
   860→            // Lock out for 5 minutes
   861→            long lockoutUntil = System.currentTimeMillis() + PIN_LOCKOUT_DURATION_MS;
   862→            getPrefs().edit()
   863→                .putLong(KEY_PIN_LOCKOUT_UNTIL, lockoutUntil)
   864→                .putInt(KEY_PIN_FAILED_ATTEMPTS, 0)
   865→                .apply();
   866→            Toast.makeText(this, "Too many attempts. Locked for 5 minutes.", Toast.LENGTH_LONG).show();
   867→        } else {
   868→            int remaining = MAX_PIN_ATTEMPTS - attempts;
   869→            new AlertDialog.Builder(this)
   870→                .setTitle("Incorrect Code")
   871→                .setMessage("The code you entered is incorrect.\n" + remaining + " attempt(s) remaining.")
   872→                .setPositiveButton("OK", null)
   873→                .show();
   874→        }
   875→    }

   882→    }

  1285→    /**
  1286→     * Checks for updates in the background (no UI).
  1287→     * If an update is found and forceUpdate is true, shows a blocking dialog.
  1288→     * Otherwise, stores the info for the admin to install later.
  1289→     */
  1290→    private void checkForUpdatesInBackground() {
  1291→        new Thread(() -> {
  1292→            try {
  1293→                String baseUrl = getKioskUrl();
  1294→                // Extract origin from kiosk URL
  1295→                String origin = baseUrl;
  1296→                if (baseUrl.contains("//")) {
  1297→                    int pathStart = baseUrl.indexOf("//", baseUrl.indexOf("//") + 1);
  1298→                    if (pathStart > 0) {
  1299→                        // Find the first / after the host
  1300→                        int slashIndex = baseUrl.indexOf("/", baseUrl.indexOf("//") + 2);
  1301→                        if (slashIndex > 0) {
  1302→                            origin = baseUrl.substring(0, slashIndex);
  1303→                        } else {
  1304→                            origin = baseUrl;
  1305→                        }
  1306→                    }
  1307→                }
  1308→                String checkUrl = origin + UPDATE_CHECK_URL;
  1309→                Log.i(TAG, "Checking for updates: " + checkUrl);
  1310→
  1311→                URL url = new URL(checkUrl);
  1312→                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
  1313→                conn.setConnectTimeout(15000);
  1314→                conn.setReadTimeout(15000);
  1315→                conn.setRequestMethod("GET");
  1316→                conn.setRequestProperty("User-Agent", "PickupKiosk/" + APP_VERSION);
  1317→
  1318→                int responseCode = conn.getResponseCode();
  1319→                if (responseCode != 200) {
  1320→                    Log.w(TAG, "Update check failed: HTTP " + responseCode);
  1321→                    return;
  1322→                }
  1323→
  1324→                BufferedReader reader = new BufferedReader(new InputStreamReader(conn.getInputStream()));
  1325→                StringBuilder response = new StringBuilder();
  1326→                String line;
  1327→                while ((line = reader.readLine()) != null) {
  1328→                    response.append(line);
  1329→                }
  1330→                reader.close();
  1331→
  1332→                String jsonResponse = response.toString();
  1333→                Log.i(TAG, "Update check response: " + jsonResponse);
  1334→
  1335→                // Parse JSON manually (no Gson/Jackson in bare build)
  1336→                String serverVersion = extractJsonString(jsonResponse, "version");
  1337→                int serverVersionCode = extractJsonInt(jsonResponse, "versionCode");
  1338→                String apkUrl = extractJsonString(jsonResponse, "apkUrl");
  1339→                String checksum = extractJsonString(jsonResponse, "checksum");
  1340→                String changelog = extractJsonString(jsonResponse, "changelog");
  1341→                boolean forceUpdate = extractJsonBoolean(jsonResponse, "forceUpdate");
  1342→
  1343→                if (serverVersion == null || serverVersion.isEmpty()) {
  1344→                    Log.w(TAG, "Update check: no version info in response");
  1345→                    return;
  1346→                }
  1347→
  1348→                // Store update info
  1349→                getPrefs().edit()
  1350→                    .putString(KEY_AVAILABLE_VERSION, serverVersion)
  1351→                    .putInt(KEY_AVAILABLE_VERSION_CODE, serverVersionCode)
  1352→                    .putString(KEY_AVAILABLE_APK_URL, apkUrl)
  1353→                    .putString(KEY_AVAILABLE_CHECKSUM, checksum)
  1354→                    .putString(KEY_AVAILABLE_CHANGELOG, changelog)
  1355→                    .putBoolean(KEY_FORCE_UPDATE, forceUpdate)
  1356→                    .putLong(KEY_LAST_UPDATE_CHECK, System.currentTimeMillis())
  1357→                    .apply();
  1358→
  1359→                // Compare versions
  1360→                if (serverVersionCode > APP_VERSION_CODE) {
  1361→                    pendingUpdateVersion = serverVersion;
  1362→                    Log.i(TAG, "Update available: v" + serverVersion + " (code " + serverVersionCode + "), force=" + forceUpdate);
  1363→
  1364→                    // If force update, show blocking dialog on main thread
  1365→                    if (forceUpdate && mainHandler != null) {
  1366→                        mainHandler.post(() -> {
  1367→                            if (!isDestroyed) showForceUpdateDialog(serverVersion, changelog);
  1368→                        });
  1369→                    }
  1370→                } else {
  1371→                    pendingUpdateVersion = null;
  1372→                    Log.i(TAG, "App is up to date (v" + APP_VERSION + ")");
  1373→                }
  1374→
  1375→            } catch (Throwable t) {
  1376→                Log.e(TAG, "Update check error: " + t.getMessage());
  1377→            }
  1378→        }).start();
  1379→    }

  1570→    /**
  1571→     * Downloads the APK update and triggers installation.
  1572→     * Shows a progress dialog during download.
  1573→     */
  1574→    private void downloadAndInstallUpdate() {
  1575→        if (isDownloadingUpdate) {
  1576→            Toast.makeText(this, "Update is already downloading...", Toast.LENGTH_SHORT).show();
  1577→            return;
  1578→        }
  1579→
  1580→        String apkUrl = getPrefs().getString(KEY_AVAILABLE_APK_URL, "");
  1581→        if (apkUrl == null || apkUrl.isEmpty()) {
  1582→            // Construct download URL from server
  1583→            String baseUrl = getKioskUrl();
  1584→            String origin = baseUrl;
  1585→            if (baseUrl.contains("//")) {
  1586→                int slashIndex = baseUrl.indexOf("/", baseUrl.indexOf("//") + 2);
  1587→                if (slashIndex > 0) {
  1588→                    origin = baseUrl.substring(0, slashIndex);
  1589→                }
  1590→            }
  1591→            apkUrl = origin + "/api/app-version/download";
  1592→        }
  1593→
  1594→        final String downloadUrl = apkUrl;
  1595→        isDownloadingUpdate = true;
  1596→
  1597→        ProgressDialog progressDialog = new ProgressDialog(this);
  1598→        progressDialog.setTitle("Downloading Update");
  1599→        progressDialog.setMessage("Preparing download...");
  1600→        progressDialog.setProgressStyle(ProgressDialog.STYLE_HORIZONTAL);
  1601→        progressDialog.setMax(100);
  1602→        progressDialog.setProgress(0);
  1603→        progressDialog.setCancelable(false);
  1604→        progressDialog.show();
  1605→
  1606→        new Thread(() -> {
  1607→            try {
  1608→                // Create update directory
  1609→                File updateDir = new File(getExternalFilesDir(null), UPDATE_DIR);
  1610→                if (!updateDir.exists()) {
  1611→                    updateDir.mkdirs();
  1612→                }
  1613→                File apkFile = new File(updateDir, APK_FILENAME);
  1614→
  1615→                // Delete any previous download
  1616→                if (apkFile.exists()) {
  1617→                    apkFile.delete();
  1618→                }
  1619→
  1620→                Log.i(TAG, "Downloading update from: " + downloadUrl);
  1621→
  1622→                URL url = new URL(downloadUrl);
  1623→                HttpURLConnection conn = (HttpURLConnection) url.openConnection();
  1624→                conn.setConnectTimeout(30000);
  1625→                conn.setReadTimeout(120000); // 2 min read timeout for large APK
  1626→                conn.setRequestMethod("GET");
  1627→                conn.setRequestProperty("User-Agent", "PickupKiosk/" + APP_VERSION);
  1628→
  1629→                int responseCode = conn.getResponseCode();
  1630→                if (responseCode != 200) {
  1631→                    mainHandler.post(() -> {
  1632→                        progressDialog.dismiss();
  1633→                        isDownloadingUpdate = false;
  1634→                        Toast.makeText(this, "Download failed (HTTP " + responseCode + ")", Toast.LENGTH_LONG).show();
  1635→                    });
  1636→                    return;
  1637→                }
  1638→
  1639→                int contentLength = conn.getContentLength();
  1640→                InputStream input = conn.getInputStream();
  1641→                FileOutputStream output = new FileOutputStream(apkFile);
  1642→
  1643→                byte[] buffer = new byte[8192];
  1644→                long totalRead = 0;
  1645→                int bytesRead;
  1646→
  1647→                while ((bytesRead = input.read(buffer)) != -1) {
  1648→                    output.write(buffer, 0, bytesRead);
  1649→                    totalRead += bytesRead;
  1650→
  1651→                    if (contentLength > 0) {
  1652→                        int progress = (int) ((totalRead * 100) / contentLength);
  1653→                        final int p = progress;
  1654→                        final long mb = totalRead / (1024 * 1024);
  1655→                        mainHandler.post(() -> {
  1656→                            if (!isDestroyed) {
  1657→                                progressDialog.setProgress(p);
  1658→                                progressDialog.setMessage("Downloaded: " + mb + " MB");
  1659→                            }
  1660→                        });
  1661→                    }
  1662→                }
  1663→
  1664→                output.flush();
  1665→                output.close();
  1666→                input.close();
  1667→
  1668→                // Verify checksum if available
  1669→                String expectedChecksum = getPrefs().getString(KEY_AVAILABLE_CHECKSUM, "");
  1670→                if (expectedChecksum != null && !expectedChecksum.isEmpty() && expectedChecksum.startsWith("sha256:")) {
  1671→                    String expectedHash = expectedChecksum.substring(7);
  1672→                    String actualHash = computeSHA256(apkFile);
  1673→                    if (!expectedHash.equalsIgnoreCase(actualHash)) {
  1674→                        Log.e(TAG, "Checksum mismatch! Expected: " + expectedHash + " Got: " + actualHash);
  1675→                        apkFile.delete();
  1676→                        mainHandler.post(() -> {
  1677→                            progressDialog.dismiss();
  1678→                            isDownloadingUpdate = false;
  1679→                            new AlertDialog.Builder(this)
  1680→                                .setTitle("Update Failed")
  1681→                                .setMessage("Downloaded file is corrupted (checksum mismatch). Please try again.")
  1682→                                .setPositiveButton("OK", null)
  1683→                                .show();
  1684→                        });
  1685→                        return;
  1686→                    }
  1687→                    Log.i(TAG, "Checksum verified OK");
  1688→                }
  1689→
  1690→                // Download complete — install
  1691→                mainHandler.post(() -> {
  1692→                    if (isDestroyed) return;
  1693→                    progressDialog.dismiss();
  1694→                    isDownloadingUpdate = false;
  1695→                    installApk(apkFile);
  1696→                });
  1697→
  1698→            } catch (Throwable t) {
  1699→                Log.e(TAG, "Download failed: " + t.getMessage(), t);

  1779→    /**
  1780→     * Computes SHA-256 hash of a file.
  1781→     */
  1782→    private String computeSHA256(File file) throws Exception {
  1783→        MessageDigest digest = MessageDigest.getInstance("SHA-256");
  1784→        FileInputStream fis = new FileInputStream(file);
  1785→        byte[] buffer = new byte[8192];
  1786→        int bytesRead;
  1787→        while ((bytesRead = fis.read(buffer)) != -1) {
  1788→            digest.update(buffer, 0, bytesRead);
  1789→        }
  1790→        fis.close();
  1791→
  1792→        byte[] hash = digest.digest();
  1793→        StringBuilder hexString = new StringBuilder();
  1794→        for (byte b : hash) {
  1795→            String hex = Integer.toHexString(0xff & b);
  1796→            if (hex.length() == 1) hexString.append('0');
  1797→            hexString.append(hex);
  1798→        }
  1799→        return hexString.toString();
  1800→    }

  1802→    /**
  1803→     * Simple JSON string extractor (no external JSON library).
  1804→     * Handles: "key": "value"
  1805→     */
  1806→    private String extractJsonString(String json, String key) {
  1807→        try {
  1808→            String searchKey = "\"" + key + "\"";
  1809→            int keyIndex = json.indexOf(searchKey);
  1810→            if (keyIndex < 0) return "";
  1811→
  1812→            // Find the colon after the key
  1813→            int colonIndex = json.indexOf(":", keyIndex + searchKey.length());
  1814→            if (colonIndex < 0) return "";
  1815→
  1816→            // Find the opening quote of the value
  1817→            int openQuote = json.indexOf("\"", colonIndex + 1);
  1818→            if (openQuote < 0) return "";
  1819→
  1820→            // Find the closing quote
  1821→            int closeQuote = json.indexOf("\"", openQuote + 1);
  1822→            if (closeQuote < 0) return "";
  1823→
  1824→            return json.substring(openQuote + 1, closeQuote);
  1825→        } catch (Throwable t) {
  1826→            return "";
  1827→        }
  1828→    }

  1830→    /**
  1831→     * Simple JSON integer extractor.
  1832→     */
  1833→    private int extractJsonInt(String json, String key) {
  1834→        try {
  1835→            String searchKey = "\"" + key + "\"";
  1836→            int keyIndex = json.indexOf(searchKey);
  1837→            if (keyIndex < 0) return 0;
  1838→
  1839→            int colonIndex = json.indexOf(":", keyIndex + searchKey.length());
  1840→            if (colonIndex < 0) return 0;
  1841→
  1842→            // Skip whitespace
  1843→            int start = colonIndex + 1;
  1844→            while (start < json.length() && Character.isWhitespace(json.charAt(start))) start++;
  1845→
  1846→            // Find end of number
  1847→            int end = start;
  1848→            while (end < json.length() && (Character.isDigit(json.charAt(end)) || json.charAt(end) == '-')) end++;
  1849→
  1850→            if (start >= end) return 0;
  1851→            return Integer.parseInt(json.substring(start, end));
  1852→        } catch (Throwable t) {
  1853→            return 0;
  1854→        }
  1855→    }

  1858→     * Simple JSON boolean extractor.
  1859→     */
  1860→    private boolean extractJsonBoolean(String json, String key) {
  1861→        try {
  1862→            String searchKey = "\"" + key + "\"";
  1863→            int keyIndex = json.indexOf(searchKey);
  1864→            if (keyIndex < 0) return false;
  1865→
  1866→            int colonIndex = json.indexOf(":", keyIndex + searchKey.length());
  1867→            if (colonIndex < 0) return false;
  1868→
  1869→            // Check if "true" appears before the next comma or closing brace
  1870→            int start = colonIndex + 1;
  1871→            while (start < json.length() && Character.isWhitespace(json.charAt(start))) start++;
  1872→
  1873→            return json.substring(start, Math.min(start + 4, json.length())).equals("true");
  1874→        } catch (Throwable t) {
  1875→            return false;
  1876→        }
