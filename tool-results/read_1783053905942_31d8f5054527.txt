   463→const KIOSK_JS = `
   464→  // Inactivity timeout - return to home after 90 seconds of no interaction
   465→  var inactivityTimer;
   466→  var INACTIVITY_MS = 90000;
   467→  var countdownTimer;
   468→  var remainingSeconds = Math.floor(INACTIVITY_MS / 1000);
   469→
   470→  function resetInactivityTimer() {
   471→    clearTimeout(inactivityTimer);
   472→    clearInterval(countdownTimer);
   473→    remainingSeconds = Math.floor(INACTIVITY_MS / 1000);
   474→    updateCountdown();
   475→    inactivityTimer = setTimeout(function() {
   476→      window.location.href = '/kiosk-lite';
   477→    }, INACTIVITY_MS);
   478→    countdownTimer = setInterval(function() {
   479→      remainingSeconds = remainingSeconds - 1;
   480→      if (remainingSeconds <= 0) {
   481→        clearInterval(countdownTimer);
   482→        return;
   483→      }
   484→      updateCountdown();
   485→    }, 1000);
   486→  }
   487→
   488→  function updateCountdown() {
   489→    var el = document.getElementById('kiosk-countdown');
   490→    if (!el) return;
   491→    var m = Math.floor(remainingSeconds / 60);
   492→    var s = remainingSeconds % 60;
   493→    if (s < 10) s = '0' + s;
   494→    el.textContent = m + ':' + s;
   495→    // Change color when low
   496→    if (remainingSeconds <= 15) {
   497→      el.style.color = '#ff6b6b';
   498→    } else if (remainingSeconds <= 30) {
   499→      el.style.color = '#FFD439';
   500→    }
   501→  }
   502→
   503→  // Listen for user activity
   504→  document.addEventListener('touchstart', resetInactivityTimer, false);
   505→  document.addEventListener('click', resetInactivityTimer, false);
   506→  document.addEventListener('keydown', resetInactivityTimer, false);
   507→
   508→  // Start timer on page load
   509→  resetInactivityTimer();
   510→
   511→  // Update clock display
   512→  function updateClock() {
   513→    var el = document.getElementById('kiosk-clock');
   514→    if (!el) return;
   515→    var now = new Date();
   516→    var h = now.getHours();
   517→    var m = now.getMinutes();
   518→    var ampm = h >= 12 ? 'PM' : 'AM';
   519→    h = h % 12;
   520→    if (h === 0) h = 12;
   521→    if (m < 10) m = '0' + m;
   522→    el.textContent = h + ':' + m + ' ' + ampm;
   523→  }
   524→  updateClock();
   525→  setInterval(updateClock, 30000);
   526→
   527→  // Try to enter fullscreen (works in some WebView versions)
   528→  function tryFullscreen() {
   529→    var el = document.documentElement;
   530→    if (el.requestFullscreen) {
   531→      el.requestFullscreen();
   532→    } else if (el.webkitRequestFullscreen) {
   533→      el.webkitRequestFullscreen();
   534→    }
   535→  }
   536→  // Auto-enter fullscreen on first touch

   541→  // ---- AJAX Payment Polling (Chrome 37 compatible) ----
   542→  // Replaces meta-refresh so QR code stays visible while checking payment status
   543→  function startPaymentPolling(pollUrl, checkIntervalMs) {
   544→    var pollInterval = setInterval(function() {
   545→      var xhr = new XMLHttpRequest();
   546→      xhr.open('GET', pollUrl, true);
   547→      xhr.setRequestHeader('Accept', 'application/json');
   548→      xhr.onreadystatechange = function() {
   549→        if (xhr.readyState !== 4) return;
   550→        if (xhr.status !== 200) return;
   551→        try {
   552→          var resp = JSON.parse(xhr.responseText);
   553→          if (resp && resp.status === 'completed') {
   554→            clearInterval(pollInterval);
   555→            // Payment confirmed — redirect to success page

   558→            clearInterval(pollInterval);

   560→          }
   561→        } catch(e) {
   562→          // Non-JSON response — ignore, keep polling
   563→        }
   564→      };
   565→      xhr.send();

   941→// POST handler - only for step=1 redirects
   942→export async function POST(request: NextRequest) {
   943→  const formData = await request.formData();
   944→  const flow = formData.get('flow') as string;
   945→  const step = formData.get('step') as string;
   946→
   947→  if (step === '1') {
   948→    if (flow === 'dropoff') {
   949→      return NextResponse.redirect(new URL('/kiosk-lite?action=dropoff', request.url), 303);
   950→    }
   951→    if (flow === 'pickup') {
   952→      return NextResponse.redirect(new URL('/kiosk-lite?action=pickup', request.url), 303);
   953→    }
   954→  }
   955→
   956→  return NextResponse.redirect(new URL('/kiosk-lite', request.url), 303);
   957→}
