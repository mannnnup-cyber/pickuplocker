    94→async function kioskFetch(url: string, options?: RequestInit, timeoutMs: number = FETCH_TIMEOUT_MS): Promise<Response> {
    95→  const controller = new AbortController()
    96→  const timeoutId = setTimeout(() => controller.abort(), timeoutMs)
    97→  try {
    98→    const res = await fetch(url, { ...options, signal: controller.signal })
    99→    if (!res.ok) {
   100→      throw new Error(`HTTP ${res.status} from ${url}`)
   101→    }
   102→    return res
   103→  } catch (error) {
   104→    if (error instanceof DOMException && error.name === 'AbortError') {
   105→      throw new Error(`Request to ${url} timed out after ${timeoutMs}ms`)
   106→    }
   107→    throw error
   108→  } finally {
   109→    clearTimeout(timeoutId)
   110→  }
   111→}

   629→    const maxAttempts = 45 // ~6 minutes with backoff

   631→
   632→    const poll = async () => {
   633→      // Stop if component unmounted
   634→      if (!isMountedRef.current) return

   645→          if (document.visibilityState === 'visible') {
   646→            document.removeEventListener('visibilitychange', handleVisible)
   647→            if (visibilityHandlerRef.current === handleVisible) {
   648→              visibilityHandlerRef.current = null
   649→            }
   650→            poll()
   651→          }
   652→        }
   653→        visibilityHandlerRef.current = handleVisible
   654→        document.addEventListener('visibilitychange', handleVisible)
   655→        return
   656→      }

   658→      attempts++

   661→        const res = await kioskFetch(`/api/kiosk/payment?paymentId=${paymentId}`, undefined, 10000)

   674→        if (data.status === "failed") {
   675→          setError("Payment failed. Please try again.")
   676→          return
   677→        }
   678→
   679→        // Exponential backoff: 5s for first 10 attempts, then 8s, then 10s
   680→        const delay = attempts < 10 ? 5000 : attempts < 25 ? 8000 : 10000
   681→        pollingTimerRef.current = setTimeout(poll, delay)

   683→        // Continue polling on error with longer delay
