    10→// Fetch with timeout for Bestwond API calls
    11→async function fetchWithTimeout(url: string, options: RequestInit): Promise<Response> {
    12→  const controller = new AbortController();
    13→  const timeoutId = setTimeout(() => controller.abort(), BESTWOND_TIMEOUT_MS);
    14→  try {
    15→    const response = await fetch(url, { ...options, signal: controller.signal });
    16→    return response;
    17→  } finally {
    18→    clearTimeout(timeoutId);
    19→  }
    20→}

    41→// Generate SHA512 signature for Bestwond API
    42→// Signature = SHA512(sorted_url_encoded_params + app_secret)
    43→export async function generateSignature(params: Record<string, string | number>, appSecret: string): Promise<string> {
    44→  // Sort parameters alphabetically by key
    45→  const sortedKeys = Object.keys(params).sort();
    46→  
    47→  // Build URL-encoded string
    48→  const encodedParams = sortedKeys
    49→    .map(key => `${encodeURIComponent(key)}=${encodeURIComponent(params[key])}`)
    50→    .join('&');
    51→  
    52→  // Append app secret
    53→  const stringToSign = `${encodedParams}${appSecret}`;
    54→  
    55→  // SHA512 hash
    56→  const encoder = new TextEncoder();
    57→  const dataBuffer = encoder.encode(stringToSign);
    58→  const hashBuffer = await crypto.subtle.digest('SHA-512', dataBuffer);
    59→  const hashArray = Array.from(new Uint8Array(hashBuffer));
    60→  return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
    61→}

   135→// Verify webhook signature from Bestwond
   136→export function verifyWebhookSignature(payload: string, signature: string): boolean {
   137→  const config = getConfig();
   138→  
   139→  if (!config.appSecret) {
   140→    console.error('Bestwond app secret not configured - webhook verification failed');
   141→    return false; // Reject unsigned webhooks in production
   142→  }
   143→  
   144→  // Bestwond signs webhooks with SHA512(payload + appSecret)
   145→  const expectedSignature = crypto
   146→    .createHmac('sha512', config.appSecret)
   147→    .update(payload)
   148→    .digest('hex');
   149→  
   150→  return signature === expectedSignature;
   151→}

   310→// Open box and verify door actually opened
   311→export async function openBoxAndVerify(
   312→  deviceNumber: string,
   313→  boxNo: number,
   314→  credentials: BestwondCredentials,
   315→  maxRetries: number = 3
   316→): Promise<{ success: boolean; message: string; doorStatus?: string; apiResponse?: BestwondResponse }> {
   317→  // First, check if device is online
   318→  const statusResult = await getDeviceStatusWithCredentials(deviceNumber, credentials);
   319→  if (statusResult.code !== 0) {
   320→    return { 
   321→      success: false, 
   322→      message: `Cannot reach device. API error: ${statusResult.msg || 'Unknown error'}` 
   323→    };
   324→  }
   325→  
   326→  const deviceData = statusResult.data;
   327→  const isOnline = deviceData?.status === 'on' || deviceData?.online === true;
   328→  
   329→  if (!isOnline) {
   330→    return { 
   331→      success: false, 
   332→      message: `Device ${deviceNumber} is OFFLINE. Please check power and network connection.` 
   333→    };
   334→  }
   335→  
   336→  // Get the lock_address for this box
   337→  const boxListResult = await getBoxListWithCredentials(deviceNumber, credentials);
   338→  let lockAddress = `01${boxNo.toString(16).toLowerCase().padStart(2, '0')}`; // HEX format
   339→  
   340→  if (boxListResult.code === 0 && boxListResult.data) {
   341→    const box = boxListResult.data.find(b => parseInt(b.box_name, 10) === boxNo);
   342→    if (box?.lock_address) {
   343→      lockAddress = box.lock_address;
   344→    }
   345→  }
   346→  
   347→  // Try to open the box
   348→  let lastError: string | undefined;
   349→  for (let attempt = 1; attempt <= maxRetries; attempt++) {
   350→    console.log(`Open box attempt ${attempt}/${maxRetries}`);
   351→    
   352→    const openResult = await openBoxWithCredentials(deviceNumber, boxNo, credentials);
   353→    
   354→    // Check both API response code AND device status
   355→    const resultData = openResult.data as { status?: string; msg?: string } | undefined;
   356→    const deviceStatus = resultData?.status;
   357→    const deviceMsg = resultData?.msg;
   358→    
   359→    if (openResult.code === 0 && deviceStatus === 'success') {
   360→      // API and device both say success - wait and check door
   361→      await new Promise(resolve => setTimeout(resolve, 1500));
   362→      
   363→      const doorStatus = await getDoorStatusWithCredentials(deviceNumber, lockAddress, credentials);
   364→      
   365→      return {
   366→        success: true,
   367→        message: `Box #${boxNo} opened successfully.`,
   368→        doorStatus: doorStatus.data?.status,
   369→        apiResponse: openResult
   370→      };
   371→    }
   372→    
   373→    // Check for specific device errors
   374→    if (deviceStatus === 'fail') {
   375→      // Device rejected the command - no point retrying
   376→      if (deviceMsg?.includes('uqkey') || deviceMsg?.includes('key')) {
   377→        return {
   378→          success: false,
   379→          message: `❌ DEVICE NOT LINKED: This device is not registered to your app account. Contact Bestwond support to link device ${deviceNumber} to app ${credentials.appId}. Error: ${deviceMsg}`,
   380→          apiResponse: openResult
   381→        };
   382→      }
   383→      
   384→      if (deviceMsg?.includes('offline') || deviceMsg?.includes('not online')) {
   385→        return {
   386→          success: false,
   387→          message: `❌ DEVICE OFFLINE: Device ${deviceNumber} is not connected. Check power and network.`,
   388→          apiResponse: openResult
   389→        };
   390→      }
   391→      
   392→      // Other device error
   393→      lastError = deviceMsg || 'Device rejected command';
   394→    } else {
   395→      lastError = openResult.msg || `Error code: ${openResult.code}`;
   396→    }
   397→    
   398→    console.log(`Attempt ${attempt} failed:`, lastError);
   399→    
   400→    // Wait before retry
   401→    if (attempt < maxRetries) {
   402→      await new Promise(resolve => setTimeout(resolve, 1000));
   403→    }
   404→  }
   405→  
   406→  return {
   407→    success: false,
   408→    message: `Failed to open box after ${maxRetries} attempts. Last error: ${lastError}`
   409→  };
   410→}

   585→  openResult: BestwondResponse;
   586→  doorStatus: BestwondResponse;
   587→}> {
   588→  console.log('====== COMPREHENSIVE DEBUG ======');
   589→  console.log('Device:', deviceNumber, 'Box:', boxNo);
   590→  console.log('Credentials:', credentials.appId?.substring(0, 10) + '...');
   591→  
   592→  // Run all checks in parallel where possible
   593→  const [deviceStatus, boxList, deviceList] = await Promise.all([
   594→    getDeviceStatusWithCredentials(deviceNumber, credentials),
   595→    getBoxListWithCredentials(deviceNumber, credentials),
   596→    getDeviceListWithCredentials(credentials),
   597→  ]);
   598→  
   599→  // Get box log
   600→  const boxLog = await getBoxLogWithCredentials(deviceNumber, credentials, { boxNo, pageSize: 5 });
   601→  
   602→  // Try to open the box
   603→  const openResult = await openBoxWithCredentials(deviceNumber, boxNo, credentials);
   604→  
   605→  // Wait a moment then check door status
   606→  await new Promise(resolve => setTimeout(resolve, 2000));
   607→  
   608→  // Get lock address from box list
   609→  let lockAddress = `01${boxNo.toString(16).toLowerCase().padStart(2, '0')}`;
   610→  if (boxList.code === 0 && boxList.data) {
   611→    const box = (boxList.data as BoxInfo[]).find(b => parseInt(b.box_name, 10) === boxNo);
   612→    if (box?.lock_address) {
   613→      lockAddress = box.lock_address;
   614→    }
   615→  }
   616→  
   617→  const doorStatus = await getDoorStatusWithCredentials(deviceNumber, lockAddress, credentials);
   618→  
   619→  // Get updated box log after open attempt
   620→  const updatedBoxLog = await getBoxLogWithCredentials(deviceNumber, credentials, { boxNo, pageSize: 5 });
   621→  
   622→  console.log('====== END DEBUG ======');
   623→  
   624→  return {
   625→    deviceStatus,
   626→    boxList,
   627→    deviceList,
   628→    boxLog: updatedBoxLog,
   629→    openResult,
   630→    doorStatus,
   631→  };
   632→}
