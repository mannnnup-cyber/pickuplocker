   237→                break;
   238→            default:
   239→                setRequestedOrientation(android.content.pm.ActivityInfo.SCREEN_ORIENTATION_SENSOR_PORTRAIT);

   241→        }

   357→    private void configureWebView() {
   358→        if (webView == null) return;
   359→        try {
   360→            WebSettings settings = webView.getSettings();
   361→            settings.setJavaScriptEnabled(true);
   362→            settings.setDomStorageEnabled(true);
   363→            settings.setAllowFileAccess(true);
   364→            settings.setAllowContentAccess(true);
   365→            if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.LOLLIPOP) {
   366→                settings.setMixedContentMode(WebSettings.MIXED_CONTENT_ALWAYS_ALLOW);
   367→            }
   368→            settings.setCacheMode(WebSettings.LOAD_DEFAULT);
   369→            settings.setBuiltInZoomControls(false);
   370→            settings.setDisplayZoomControls(false);
   371→
   372→            String userAgent = settings.getUserAgentString();
   373→            settings.setUserAgentString(userAgent + " PickupKiosk/" + APP_VERSION);
   374→
   375→            try {
   376→                boolean isDebug = false;
   377→                try { isDebug = BuildConfig.DEBUG; } catch (Throwable ignored) {}
   378→                WebView.setWebContentsDebuggingEnabled(isDebug);
   379→            } catch (Throwable t) { Log.w(TAG, "WebView debugging: " + t.getMessage()); }
   380→
   381→            webView.setWebViewClient(new KioskWebViewClient());
   382→            webView.setWebChromeClient(new KioskWebChromeClient());
   383→            webView.loadUrl(getKioskUrl());
   384→        } catch (Throwable t) {
   385→            Log.e(TAG, "WebView config failed: " + t.getMessage(), t);
   386→            showErrorScreen("WebView configuration failed: " + t.getMessage());
   387→        }
   388→    }

   394→    private class KioskWebViewClient extends WebViewClient {
   395→        @Override
   396→        public void onPageFinished(WebView view, String url) {
   397→            super.onPageFinished(view, url);
   398→            if (url == null) return;
   399→            if (OFFLINE_URL.equals(url)) {
   400→                isShowingOfflinePage = true;
   401→                Log.w(TAG, "Showing offline page");
   402→            } else {
   403→                isShowingOfflinePage = false;
   404→                reconnectAttempts = 0;
   405→                healthCheckFailCount = 0; // Reset health on successful page load
   406→                Log.i(TAG, "Page loaded: " + url);
   407→            }
   408→        }
   409→
   410→        @Override
   411→        public void onReceivedError(WebView view, WebResourceRequest request, WebResourceError error) {
   412→            if (request != null && request.isForMainFrame()) {
   413→                if (!isShowingOfflinePage) showOfflinePage();
   414→            }
   415→        }
   416→
   417→        @Override
   418→        public void onReceivedError(WebView view, int errorCode, String description, String failingUrl) {
   419→            if (!isShowingOfflinePage) showOfflinePage();
   420→        }
   421→
   422→        @Override
   423→        public boolean shouldOverrideUrlLoading(WebView view, WebResourceRequest request) {
   424→            if (request == null) return false;
   425→            String url = request.getUrl().toString();
   426→            if (url == null) return true;
   427→            // Allow kiosk domain, backend, assets, and payment API
   428→            if (url.startsWith("https://pickuplocker.vercel.app") ||
   429→                url.startsWith("file:///android_asset/") ||
   430→                url.startsWith("https://api.dimepay.app")) {
   431→                return false;
   432→            }
   433→            Log.w(TAG, "Blocked navigation to: " + url);
   434→            return true;
   435→        }
   436→
   437→        /**
   438→         * CRITICAL: Called when the WebView renderer process crashes (OOM kill, etc.)
   439→         * This is the #1 cause of white screens — Android kills the renderer but
   440→         * the Activity is still alive. Without this handler, the WebView shows blank.
   441→         */
   442→        @Override
   443→        public boolean onRenderProcessGone(WebView view, android.webkit.RenderProcessGoneDetail detail) {
   444→            Log.e(TAG, "WebView renderer process GONE! Did crash: " + detail.didCrash()
   445→                + ". Rebuilding WebView immediately.");
   446→
   447→            if (isDestroyed) return true;
   448→
   449→            // Destroy the old WebView completely
   450→            if (webView != null) {
   451→                try {
   452→                    webView.stopLoading();
   453→                    webView.loadUrl("about:blank");
   454→                    webView.clearCache(true);
   455→                    webView.clearHistory();
   456→                    ViewGroup parent = (ViewGroup) webView.getParent();
   457→                    if (parent != null) parent.removeView(webView);
   458→                    webView.destroy();
   459→                } catch (Throwable t) {
   460→                    Log.e(TAG, "Error destroying dead WebView: " + t.getMessage());
   461→                }
   462→                webView = null;
   463→            }
   464→
   465→            // Rebuild a fresh WebView
   466→            mainHandler.post(() -> {
   467→                if (isDestroyed) return;
   468→                try {
   469→                    webView = createWebViewSafely();
   470→                    if (webView != null) {
   471→                        setContentView(webView);
   472→                        hideSystemUI();
   473→                        configureWebView();
   474→                        webView.loadUrl(getKioskUrl());
   475→                        isShowingOfflinePage = false;
   476→                        reconnectAttempts = 0;
   477→                        healthCheckFailCount = 0;
   478→                        Log.i(TAG, "WebView rebuilt after renderer crash — kiosk reloaded");
   479→                    } else {
   480→                        showErrorScreen("WebView failed to restart. Please reboot the device.");
   481→                    }
   482→                } catch (Throwable t) {
   483→                    Log.e(TAG, "Failed to rebuild WebView: " + t.getMessage());
   484→                    showErrorScreen("Critical error. Please reboot the device.");
   485→                }
   486→            });
   487→
   488→            return true; // We handled it — don't let WebView do its default (which does nothing)
   489→        }
   490→    }

   509→    // ============================================
   510→    // OFFLINE & RECONNECT
   511→    // ============================================
   512→
   513→    private void showOfflinePage() {
   514→        isShowingOfflinePage = true;
   515→        if (webView != null && mainHandler != null) {
   516→            mainHandler.post(() -> {
   517→                if (webView != null && !isDestroyed) {
   518→                    try { webView.loadUrl(OFFLINE_URL); } catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   519→                }
   520→            });
   521→        }
   522→        scheduleReconnect();
   523→    }
   524→
   525→    private void scheduleReconnect() {
   526→        if (reconnectAttempts >= MAX_RECONNECT_ATTEMPTS) return;
   527→        reconnectAttempts++;
   528→        if (mainHandler != null) {
   529→            mainHandler.postDelayed(() -> {
   530→                if (isShowingOfflinePage && !isDestroyed) checkAndReload();
   531→            }, RECONNECT_DELAY_MS);
   532→        }
   533→    }
   534→
   535→    private void checkAndReload() {
   536→        if (!isNetworkAvailable()) { scheduleReconnect(); return; }
   537→        if (webView != null && mainHandler != null) {
   538→            mainHandler.post(() -> {
   539→                if (webView != null && !isDestroyed) {
   540→                    try { webView.loadUrl(isViewingBackend ? getBackendUrl() : getKioskUrl()); }
   541→                    catch (Throwable t) { Log.e(TAG, t.getMessage()); }
   542→                }
   543→            });
   544→        }
   545→    }

   548→    // CONNECTIVITY MONITORING
   549→    // ============================================
   550→
   551→    private void registerConnectivityMonitoring() {
   552→        try {
   553→            ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   554→            if (cm == null) return;
   555→            NetworkRequest request = new NetworkRequest.Builder()
   556→                .addCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   557→                .build();
   558→            networkCallback = new ConnectivityManager.NetworkCallback() {
   559→                @Override
   560→                public void onAvailable(Network network) {
   561→                    if (isShowingOfflinePage && mainHandler != null) {
   562→                        mainHandler.postDelayed(() -> {
   563→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   564→                                reconnectAttempts = 0;
   565→                                mainHandler.post(() -> {
   566→                                    if (webView != null && !isDestroyed)
   567→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   568→                                });
   569→                            }
   570→                        }, CONNECTIVITY_CHECK_MS);
   571→                    }
   572→                }
   573→                @Override
   574→                public void onLost(Network network) { Log.w(TAG, "Network lost"); }
   575→                @Override
   576→                public void onCapabilitiesChanged(Network network, NetworkCapabilities caps) {
   577→                    if (caps == null) return;
   578→                    boolean ok = caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_INTERNET)
   579→                            && caps.hasCapability(NetworkCapabilities.NET_CAPABILITY_VALIDATED);
   580→                    if (ok && isShowingOfflinePage && mainHandler != null) {
   581→                        mainHandler.postDelayed(() -> {
   582→                            if (isShowingOfflinePage && webView != null && !isDestroyed) {
   583→                                reconnectAttempts = 0;
   584→                                mainHandler.post(() -> {
   585→                                    if (webView != null && !isDestroyed)
   586→                                        try { webView.loadUrl(getKioskUrl()); } catch (Throwable t) {}
   587→                                });
   588→                            }
   589→                        }, CONNECTIVITY_CHECK_MS);
   590→                    }
   591→                }
   592→            };
   593→            cm.registerNetworkCallback(request, networkCallback);
   594→        } catch (Throwable t) {
   595→            Log.e(TAG, "Network monitoring failed: " + t.getMessage());
   596→        }
   597→    }

   599→    private void unregisterConnectivityMonitoring() {
   600→        if (networkCallback != null) {
   601→            try {
   602→                ConnectivityManager cm = (ConnectivityManager) getSystemService(Context.CONNECTIVITY_SERVICE);
   603→                if (cm != null) cm.unregisterNetworkCallback(networkCallback);
   604→            } catch (Throwable t) { Log.w(TAG, t.getMessage()); }
   605→        }
   606→    }

   616→        } catch (Throwable t) { return true; }

   620→    // WEBVIEW HEALTH MONITORING — Detect frozen/white screens
   621→    // ============================================
   622→
   623→    /**
   624→     * Starts periodic health checks on the WebView.
   625→     * If the WebView appears unresponsive or has blank content for
   626→     * multiple consecutive checks, force a reload.
   627→     */
   628→    private void startHealthMonitoring() {
   629→        healthCheckRunnable = new Runnable() {
   630→            @Override
   631→            public void run() {
   632→                if (!isDestroyed) {
   633→                    checkWebViewHealth();
   634→                    mainHandler.postDelayed(this, HEALTH_CHECK_INTERVAL_MS);
   635→                }
   636→            }
   637→        };
   638→        mainHandler.postDelayed(healthCheckRunnable, HEALTH_CHECK_INTERVAL_MS);
   639→        Log.i(TAG, "WebView health monitoring started (interval: " + HEALTH_CHECK_INTERVAL_MS + "ms)");
   640→    }
   641→
   642→    private void stopHealthMonitoring() {
   643→        if (healthCheckRunnable != null) {
   644→            mainHandler.removeCallbacks(healthCheckRunnable);
   645→            healthCheckRunnable = null;
   646→            Log.i(TAG, "WebView health monitoring stopped");
   647→        }
   648→    }

   650→    /**
   651→     * Check if the WebView is healthy by evaluating its content.
   652→     * If the page appears blank or frozen for several consecutive checks,
   653→     * force a reload to recover from white screen issues.
   654→     */
   655→    private void checkWebViewHealth() {
   656→        if (webView == null) {
   657→            healthCheckFailCount++;
   658→            Log.w(TAG, "Health check: WebView is null (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   659→            if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   660→                Log.w(TAG, "Health check: Too many failures — attempting full recovery");
   661→                healthCheckFailCount = 0;
   662→                reloadKiosk();
   663→            }
   664→            return;
   665→        }
   666→
   667→        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.KITKAT) {
   668→            webView.evaluateJavascript("(function(){ try { var c = document.querySelector('.container'); return c ? 'ok' : 'empty'; } catch(e) { return 'error'; } })()", new android.webkit.ValueCallback<String>() {
   669→                @Override
   670→                public void onReceiveValue(String value) {
   671→                    if (value != null && (value.contains("ok") || value.contains("empty"))) {
   672→                        if (value.contains("empty")) {
   673→                            healthCheckFailCount++;
   674→                            Log.w(TAG, "Health check: Page content empty (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   675→                        } else {
   676→                            if (healthCheckFailCount > 0) {
   677→                                Log.i(TAG, "Health check: Page recovered (was " + healthCheckFailCount + " fails)");
   678→                            }
   679→                            healthCheckFailCount = 0;
   680→                        }
   681→                    } else {
   682→                        healthCheckFailCount++;
   683→                        Log.w(TAG, "Health check: No response or unexpected value: " + value + " (fail " + healthCheckFailCount + "/" + HEALTH_RELOAD_THRESHOLD + ")");
   684→                    }
   685→
   686→                    if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   687→                        Log.w(TAG, "Health check: Threshold reached — force reloading kiosk");
   688→                        healthCheckFailCount = 0;
   689→                        reloadKiosk();
   690→                    }
   691→                }
   692→            });
   693→        } else {
   694→            // Pre-KitKat: just check offline status
   695→            if (isShowingOfflinePage) {
   696→                healthCheckFailCount++;
   697→                if (healthCheckFailCount >= HEALTH_RELOAD_THRESHOLD) {
   698→                    healthCheckFailCount = 0;
   699→                    reloadKiosk();
   700→                }
   701→            } else {
   702→                healthCheckFailCount = 0;
   703→            }
   704→        }
   705→    }

   730→                if (x < hitAreaPx && y < hitAreaPx) {
   731→                    long now = System.currentTimeMillis();
   732→                    if (now - lastSecretTapTime > SECRET_TAP_TIMEOUT_MS) {
   733→                        secretTapCount = 1;
   734→                    } else {
   735→                        secretTapCount++;
   736→                    }
   737→                    lastSecretTapTime = now;
   738→                    if (secretTapCount >= SECRET_TAP_COUNT) {
   739→                        secretTapCount = 0;
   740→                        Log.i(TAG, "Secret trigger activated");
   741→                        showPinDialog();
   742→                    }
   743→                }
   744→            } catch (Throwable t) {
   745→                Log.w(TAG, "Touch handler error: " + t.getMessage());
   746→            }
   747→        }
   748→        return super.dispatchTouchEvent(ev);
   749→    }

   815→    private void handleWrongPin() {

   828→        } else {
   829→            int remaining = MAX_PIN_ATTEMPTS - attempts;
   830→            new AlertDialog.Builder(this)
   831→                .setTitle("Incorrect Code")
   832→                .setMessage("The code you entered is incorrect.\n" + remaining + " attempt(s) remaining.")
   833→                .setPositiveButton("OK", null)
   834→                .show();
   835→        }
   836→    }
