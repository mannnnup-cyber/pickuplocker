    18→// Generate QR code as data URL
    19→async function generateQRCodeDataUrl(data: string): Promise<string> {
    20→  try {
    21→    const qrDataUrl = await QRCode.toDataURL(data, {
    22→      width: 300,
    23→      margin: 2,
    24→      color: {
    25→        dark: '#111111',
    26→        light: '#ffffff',
    27→      },
    28→    });
    29→    return qrDataUrl;

    31→    console.error('Failed to generate QR code:', error);

    36→// POST /api/kiosk/payment - Create payment for drop-off credit or check status
    37→export async function POST(request: NextRequest) {
    38→  try {
    39→    const body = await request.json();
    40→    const { action, boxSize, phone, email } = body;
    41→
    42→    console.log('[Kiosk Payment] POST Request:', { action, boxSize });
    43→
    44→    // Create drop-off payment
    45→    if (action === 'create_dropoff_payment') {
    46→      return await createDropoffPayment(boxSize, phone, email);
    47→    }
    48→
    49→    // Open box immediately after payment (skip code entry)
    50→    if (action === 'open_box_after_payment') {
    51→      const { saveCode, recipientPhone } = body;
    52→      return await openBoxAfterPayment(saveCode, recipientPhone);
    53→    }
    54→
    55→    // Create storage fee payment
    56→    if (action === 'create_storage_fee_payment') {
    57→      const { orderId, amount } = body;
    58→      return await createStorageFeePayment(orderId, amount, phone);
    59→    }
    60→
    61→    return NextResponse.json({
    62→      success: false,
    63→      error: 'Invalid action. Use: create_dropoff_payment, create_storage_fee_payment',
    64→    }, { status: 400 });
    65→
    66→  } catch (error) {
    67→    console.error('Kiosk payment error:', error);
    68→    return NextResponse.json({
    69→      success: false,
    70→      error: error instanceof Error ? error.message : 'Failed to process payment',
    71→    }, { status: 500 });
    72→  }
    73→}

   314→// Create drop-off credit payment - Uses real DimePay if configured, otherwise demo mode
   315→async function createDropoffPayment(boxSize: string, phone: string, email?: string) {
   316→  console.log('[Kiosk Payment] createDropoffPayment called with boxSize:', boxSize);
   317→
   318→  // Validate box size
   319→  if (!boxSize || !['S', 'M', 'L', 'XL'].includes(boxSize)) {
   320→    console.log('[Kiosk Payment] Invalid box size:', boxSize);
   321→    return NextResponse.json({
   322→      success: false,
   323→      error: 'Invalid box size. Must be S, M, L, or XL',
   324→    }, { status: 400 });
   325→  }
   326→
   327→  // Validate phone
   328→  if (!phone || phone.length < 7) {
   329→    console.log('[Kiosk Payment] Invalid phone:', phone);
   330→    return NextResponse.json({
   331→      success: false,
   332→      error: 'Valid phone number is required (at least 7 digits)',
   333→    }, { status: 400 });
   334→  }
   335→
   336→  const cleanPhone = phone.replace(/[^0-9+]/g, '');
   337→  const amount = BOX_PRICES[boxSize];
   338→  const saveCode = generateTrackingCode();
   339→  const pickCode = generateTrackingCode(); // Generate pick code upfront
   340→
   341→  // Create or update customer record DURING payment creation (not just after)
   342→  let customer = await db.user.findFirst({
   343→    where: { phone: cleanPhone }
   344→  });
   345→
   346→  if (!customer) {
   347→    customer = await db.user.create({
   348→      data: {
   349→        phone: cleanPhone,
   350→        email: email || `${cleanPhone}@pickup.local`,
   351→        name: 'Customer',
   352→        role: 'CUSTOMER',
   353→      }
   354→    });
   355→    console.log('[Kiosk Payment] Created customer:', customer.id);
   356→  } else if (email && customer.email.includes('@pickup.local')) {
   357→    // Update with real email if we have one - but only if not already taken
   358→    try {
   359→      const existingWithEmail = await db.user.findFirst({
   360→        where: { email, NOT: { id: customer.id } }
   361→      });
   362→      if (!existingWithEmail) {
   363→        customer = await db.user.update({
   364→          where: { id: customer.id },
   365→          data: { email }
   366→        });
   367→        console.log('[Kiosk Payment] Updated customer email:', email);
   368→      } else {
   369→        console.log('[Kiosk Payment] Email already in use by another account, skipping update:', email);
   370→      }
   371→    } catch (emailError) {
   372→      console.error('[Kiosk Payment] Failed to update customer email:', emailError);
   373→    }
   374→  }
   375→
   376→  // Check if DimePay is configured
   377→  const config = await getDimepayConfig();
   378→
   379→  console.log('[Kiosk Payment] DimePay config loaded:', {
   380→    hasClientId: !!config.clientId,
   381→    hasSecretKey: !!config.secretKey,
   382→    sandboxMode: config.sandboxMode,
   383→    sandboxClientIdSet: !!config.sandboxClientId,
   384→    sandboxSecretKeySet: !!config.sandboxSecretKey,
   385→    liveClientIdSet: !!config.liveClientId,
   386→    liveSecretKeySet: !!config.liveSecretKey,
   387→  });
   388→
   389→  // Determine which credentials to use based on mode
   390→  const effectiveClientId = config.sandboxMode ? config.sandboxClientId : config.liveClientId;
   391→  const effectiveSecretKey = config.sandboxMode ? config.sandboxSecretKey : config.liveSecretKey;
   392→
   393→  console.log('[Kiosk Payment] Effective credentials for', config.sandboxMode ? 'SANDBOX' : 'LIVE', 'mode:', {
   394→    clientIdSet: !!effectiveClientId,
   395→    secretKeySet: !!effectiveSecretKey,
   396→    clientIdPrefix: effectiveClientId ? effectiveClientId.substring(0, 12) + '...' : '(none)',
   397→  });
   398→
   399→  // Use real DimePay SDK if credentials are configured
   400→  if (effectiveClientId && effectiveSecretKey) {
   401→    console.log('[Kiosk Payment] Using real DimePay SDK');
   402→
   403→    try {
   404→      // Always use production URL for QR codes (users scan with their phones)
   405→      const baseUrl = 'https://pickupja.com';
   406→      const orderId = `DROPOFF-${Date.now()}`;
   407→
   408→      // Create SDK config
   409→      const sdkConfig: DimePaySDKConfig = {
   410→        clientId: effectiveClientId,
   411→        secretKey: effectiveSecretKey,
   412→        sandboxMode: config.sandboxMode,
   413→        passFeeToCustomer: config.passFeeToCustomer,
   414→        passFeeToCourier: config.passFeeToCourier,
   415→        feePercentage: config.feePercentage,
   416→        fixedFee: config.fixedFee,
   417→      };
   418→
   419→      const result = await createSDKPayment({
   420→        amount: Math.round(amount * 100), // Convert to cents
   421→        currency: 'JMD',
   422→        orderId: orderId,
   423→        description: `Drop-off Credit - ${boxSize} Box`,
   424→        customerPhone: cleanPhone,
   425→        customerEmail: email || undefined,
   426→        redirectUrl: `${baseUrl}/kiosk?payment=success`,
   427→        webhookUrl: `${baseUrl}/api/webhooks/dimepay`,
   428→        passFeeToCustomer: config.passFeeToCustomer,
   429→        metadata: {
   430→          type: 'dropoff_credit',
   431→          boxSize,
   432→          saveCode,
   433→          pickCode,
   434→          customerId: customer.id,
   435→          customerPhone: cleanPhone,
   436→          customerEmail: email || undefined,
   437→        },
   438→      }, sdkConfig);
   439→
   440→      console.log('[Kiosk Payment] SDK Payment Result:', {
   441→        success: result.success,
   442→        amount: result.data?.amount,
   443→        originalAmount: result.data?.originalAmount,
   444→        feeAmount: result.data?.feeAmount,
   445→        passFeeToCustomer: config.passFeeToCustomer,
   446→        feePercentage: config.feePercentage,
   447→        fixedFee: config.fixedFee,
   448→      });
   449→
   450→      if (result.success && result.data && result.data.sdkConfig) {
   451→        // Generate payment URL for the payment page
   452→        const paymentUrl = `${baseUrl}/pay/${orderId}`;
   453→
   454→        // Generate QR code from payment URL
   455→        let qrCodeDataUrl: string | undefined;
   456→        try {
   457→          qrCodeDataUrl = await generateQRCodeDataUrl(paymentUrl);
   458→          console.log('[Kiosk Payment] QR code generated for payment URL');
   459→        } catch (qrError) {
   460→          console.error('[Kiosk Payment] QR generation error:', qrError);
   461→        }
   462→
   463→        // Store payment data in DATABASE for retrieval by payment page
   464→        const paymentSessionData = {
   465→          amount: result.data.amount,
   466→          originalAmount: result.data.originalAmount,
   467→          feeAmount: result.data.feeAmount,
   468→          currency: result.data.currency,
   469→          description: `Drop-off Credit - ${boxSize} Box`,
   470→          reference: orderId,
   471→          sdkConfig: result.data.sdkConfig,
   472→          metadata: {
   473→            type: 'dropoff_credit',
   474→            boxSize,
   475→            saveCode,
   476→            pickCode,
   477→            customerId: customer.id,
   478→            customerPhone: cleanPhone,
   479→            customerEmail: email || undefined,
   480→          },
   481→          status: 'PENDING',
   482→          createdAt: Date.now(),
   483→        };
   484→
   485→        await db.setting.upsert({
   486→          where: { key: `sdk_payment_${orderId}` },
   487→          create: {
   488→            key: `sdk_payment_${orderId}`,
   489→            value: JSON.stringify(paymentSessionData),
   490→            description: `SDK Payment: ${orderId}`,
   491→          },
   492→          update: {
   493→            value: JSON.stringify(paymentSessionData),
   494→          }
   495→        });
   496→
   497→        console.log('[Kiosk Payment] SDK payment stored in database:', orderId);
   498→
   499→        return NextResponse.json({
   500→          success: true,
   501→          paymentId: orderId,
   502→          paymentUrl,
   503→          qrCodeDataUrl,
   504→          amount: result.data.amount,
   505→          originalAmount: result.data.originalAmount,
   506→          feeAmount: result.data.feeAmount,
   507→          boxSize,
   508→          saveCode,
   509→          pickCode,
   510→          customerId: customer.id,
   511→          message: `Scan QR to pay JMD $${result.data.amount} for ${boxSize} box drop-off`,
   512→          isDemoMode: false,
   513→        });
   514→      } else {
   515→        console.error('[Kiosk Payment] DimePay SDK payment failed:', result.error);
   516→        // Fall through to demo mode
   517→      }
   518→    } catch (dimepayError) {
   519→      console.error('[Kiosk Payment] DimePay SDK error:', dimepayError);
   520→      // Fall through to demo mode
   521→    }
   522→  }
   523→
   524→  // Demo mode fallback
   525→  console.log('[Kiosk Payment] Using demo mode - no valid DimePay credentials');
   526→  const paymentId = `DEMO-${Date.now()}-${Math.random().toString(36).substr(2, 6).toUpperCase()}`;
   527→  const demoPaymentUrl = `https://pickupja.com/pay/${paymentId}`;

   533→    console.log('[Kiosk Payment] Demo QR code generated successfully');
   534→  } catch (qrError) {
   535→    console.error('[Kiosk Payment] QR generation error:', qrError);
   536→  }
   537→

   539→  await db.setting.upsert({
   540→    where: { key: `demo_payment_${paymentId}` },
   541→    create: {
   542→      key: `demo_payment_${paymentId}`,
   543→      value: JSON.stringify({
   544→        saveCode,
   545→        pickCode,
   546→        boxSize,
   547→        phone: cleanPhone,
   548→        email: email || undefined,
   549→        customerId: customer.id,
   550→        amount,

   552→        status: 'PENDING',

   554→      description: `Demo Payment: ${paymentId}`,
   555→    },
   556→    update: {
   557→      value: JSON.stringify({
   558→        saveCode,

   560→        boxSize,
   561→        phone: cleanPhone,

   597→  }
   598→
   599→  // Get order details for better payment context

   601→    where: { id: orderId },
   602→  });
   603→
   604→  const customerPhone = phone || order?.customerPhone || '';
   605→  const cleanPhone = customerPhone.replace(/[^0-9+]/g, '');

   614→    hasClientId: !!effectiveClientId,
   615→    hasSecretKey: !!effectiveSecretKey,
   616→  });

   622→    try {
   623→      const baseUrl = process.env.NEXT_PUBLIC_APP_URL || 'https://pickupja.com';

   637→        amount: Math.round(amount * 100), // Convert to cents
   638→        currency: 'JMD',

   641→        customerPhone: cleanPhone || undefined,
   642→        customerEmail: order?.customerEmail || undefined,
   643→        redirectUrl: `${baseUrl}/kiosk?payment=success`,

   647→          type: 'storage_fee',
   648→          orderId,

   656→        amount: result.data?.amount,
   657→      });

   661→

   774→// Open box immediately after payment - skips code entry step
   775→async function openBoxAfterPayment(saveCode: string, recipientPhone: string) {
   776→  console.log('[Kiosk Payment] openBoxAfterPayment called');
   777→
   778→  if (!saveCode || saveCode.length !== 6) {
   779→    return NextResponse.json({
   780→      success: false,
   781→      error: 'Invalid save code',
   782→    }, { status: 400 });
   783→  }
   784→
   785→  // Find the payment session with this save code
   786→  const settings = await db.setting.findMany({
   787→    where: {
   788→      OR: [
   789→        { key: { startsWith: 'sdk_payment_' } },
   790→        { key: { startsWith: 'demo_payment_' } },
   791→      ]
   792→    }
   793→  });
   794→
   795→  let paymentData: any = null;
   796→  let settingKey = '';
   797→
   798→  for (const setting of settings) {
   799→    const data = JSON.parse(setting.value);
   800→    if (data.saveCode === saveCode || data.metadata?.saveCode === saveCode) {
   801→      paymentData = data;
   802→      settingKey = setting.key;
   803→      break;
   804→    }
   805→  }
   806→
   807→  if (!paymentData) {
   808→    return NextResponse.json({
   809→      success: false,
   810→      error: 'Payment not found for this save code',
   811→    }, { status: 404 });
   812→  }
   813→
   814→  // Check payment is completed
   815→  if (paymentData.status !== 'COMPLETED') {
   816→    return NextResponse.json({
   817→      success: false,
   818→      error: 'Payment not yet completed',
   819→    }, { status: 400 });
   820→  }
   821→
   822→  // Get box info
   823→  const boxSize = paymentData.boxSize || paymentData.metadata?.boxSize;
   824→  const customerId = paymentData.customerId || paymentData.metadata?.customerId;
   825→  const customerPhone = recipientPhone || paymentData.phone || paymentData.metadata?.customerPhone;
   826→  const pickCode = paymentData.pickCode || paymentData.metadata?.pickCode || generateTrackingCode();
   827→
   828→  // Get device and assign box
   829→  const device = await db.device.findFirst({
   830→    where: { status: 'ONLINE' },
   831→    include: {
   832→      boxes: {
   833→        where: { status: 'AVAILABLE' },
   834→        take: 1,
   835→      }
   836→    }
   837→  });
   838→
   839→  if (!device || !device.boxes[0]) {
   840→    return NextResponse.json({
   841→      success: false,
   842→      error: 'No available locker boxes',
   843→    }, { status: 400 });
   844→  }
   845→
   846→  const box = device.boxes[0];
   847→  const boxName = box.boxName || box.boxNumber?.toString().padStart(2, '0') || '01';
   848→
   849→  // Generate order number
   850→  const orderNumber = `P${Date.now().toString().slice(-8)}`;
   851→
   852→  // Create ExpressOrder
   853→  const expressOrder = await db.expressOrder.create({
   854→    data: {
   855→      orderNo: orderNumber,
   856→      deviceId: device.deviceId,
   857→      boxName,
   858→      boxSize,
   859→      saveCode,
   860→      pickCode,
   861→      status: 'STORED',
   862→      customerPhone,
   863→      saveTime: new Date(),
   864→    }
   865→  });
   866→
   867→  // Update box status
   868→  await db.box.update({
   869→    where: { id: box.id },
   870→    data: { status: 'OCCUPIED', lastUsedAt: new Date() }
   871→  });
   872→
   873→  // Open the box via Bestwond
   874→  let boxOpened = false;
   875→  try {
   876→    const { getCredentialsForDevice, expressSaveOrTakeWithCredentials } = await import('@/lib/bestwond');
   877→    const credentials = await getCredentialsForDevice(device.id);
   878→    const result = await expressSaveOrTakeWithCredentials(
   879→      device.deviceId,
   880→      boxSize as 'S' | 'M' | 'L' | 'XL',
   881→      saveCode,
   882→      'save',
   883→      credentials
   884→    );
   885→    boxOpened = result.code === 0;
   886→    console.log('[Kiosk Payment] Box open result:', result);
   887→  } catch (error) {
   888→    console.error('[Kiosk Payment] Failed to open box:', error);
   889→  }
   890→
   891→  console.log('[Kiosk Payment] Box opened successfully:', { orderNumber, boxName, boxOpened });
   892→
   893→  return NextResponse.json({
   894→    success: true,
   895→    orderNo: orderNumber,
   896→    boxName,
   897→    boxOpened,
   898→    pickCode,
   899→    message: 'Locker opened! Please place your package inside and close the door.',
   900→  });
   901→}
